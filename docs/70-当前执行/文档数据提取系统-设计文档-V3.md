## 一、数据库设计（简化版）

### 1.1 文档提取规则主表 `cert_doc_extraction_rule`

```sql
CREATE TABLE cert_doc_extraction_rule (
    Id BIGINT PRIMARY KEY AUTO_INCREMENT COMMENT '主键ID',
    
    -- 规则编码
    RuleCode VARCHAR(50) NOT NULL COMMENT '规则编码: RULE-{序号}',
    
    -- 关联文档（核心）
    FileCode VARCHAR(200) NOT NULL COMMENT '标准文档编码（关联cert_standard_file）',
    
    -- AI提示词（核心资产）
    Prompt TEXT NOT NULL COMMENT 'AI提取提示词（包含字段/表格定义+输出格式要求）',
    
    -- 状态
    Status TINYINT DEFAULT 1 COMMENT '状态: 0禁用 1启用',
    
    -- 审计字段
    CreateDate DATETIME DEFAULT CURRENT_TIMESTAMP,
    CreateID BIGINT,
    Creator VARCHAR(50),
    ModifyDate DATETIME,
    ModifyID BIGINT,
    Modifier VARCHAR(50),
    
    UNIQUE KEY UK_RuleCode (RuleCode),
    UNIQUE KEY UK_FileCode (FileCode), -- 一个文档一个规则
    INDEX IX_Status (Status)
    
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='文档提取规则主表（核心: Prompt提示词）';
```

**设计说明**:
- 核心字段只有3个: `FileCode`(关联文档), `Prompt`(提示词), `Status`(状态)
- 文档类型、标准、阶段等信息通过`FileCode`关联`cert_standard_file`表获取
- AI模型是全局配置，不在这里存储
- 提取方式、验证规则等都在`Prompt`中定义

```sql
    -- 显示顺序
    SortOrder INT DEFAULT 0 COMMENT '显示顺序',
    
    -- 状态

    -- 审计字段
    CreateDate DATETIME DEFAULT CURRENT_TIMESTAMP,
    CreateID BIGINT,
    Creator VARCHAR(50),
    ModifyDate DATETIME,
    ModifyID BIGINT,
    Modifier VARCHAR(50),
```

- **不存AI置信度**（提取时动态计算）

### 2.2 配置读取服务

```csharp
public class QwenAIConfigService
{
    private readonly IRepository<SysParameter> _paramRepository;
    private readonly IConfiguration _configuration;
    
    public QwenAIConfig GetConfig()
    {
        // 优先从sys_parameter表读取
        var config = TryGetFromSysParameter();
        if (config != null) return config;
        
        // 否则从appsettings读取
        return _configuration.GetSection("QwenAI").Get<QwenAIConfig>();
    }
    
    private QwenAIConfig TryGetFromSysParameter()
    {
        try
        {
            var params = _paramRepository.Find(
                p => p.ParamKey.StartsWith("AI_Qwen_")
            ).ToList();
            
            if (!params.Any()) return null;
            
            return new QwenAIConfig
            {
                AccessKeyId = params.FirstOrDefault(p => p.ParamKey == "AI_Qwen_AccessKeyId")?.ParamValue,
                AccessKeySecret = params.FirstOrDefault(p => p.ParamKey == "AI_Qwen_AccessKeySecret")?.ParamValue,
                Model = params.FirstOrDefault(p => p.ParamKey == "AI_Qwen_Model")?.ParamValue ?? "qwen-turbo",
                Endpoint = params.FirstOrDefault(p => p.ParamKey == "AI_Qwen_Endpoint")?.ParamValue,
                MaxTokens = int.Parse(params.FirstOrDefault(p => p.ParamKey == "AI_Qwen_MaxTokens")?.ParamValue ?? "2000"),
                Temperature = double.Parse(params.FirstOrDefault(p => p.ParamKey == "AI_Qwen_Temperature")?.ParamValue ?? "0.3")
            };
        }
        catch
        {
            return null;
        }
    }
}
```

## 三、核心业务逻辑

### 3.1 提示词生成流程

```csharp
public class PromptGenerationService
{
    /// <summary>
    /// 生成提取提示词（核心方法）
    /// </summary>
    public async Task<string> GeneratePromptAsync(long ruleId)
    {
        // 1. 获取规则信息
        var rule = await _ruleRepository.GetAsync(ruleId);
        
        // 2. 获取字段定义列表
        var fields = await _fieldRepository.GetByRuleIdAsync(ruleId);
        
        // 3. 获取表格定义列表（含表格字段）
        var tables = await _tableRepository.GetByRuleIdWithFieldsAsync(ruleId);
        
        // 4. 构建提示词
        var prompt = BuildPrompt(fields, tables);
        
        return prompt;
    }
    
    private string BuildPrompt(List<FieldDef> fields, List<TableDef> tables)
    {
        var sb = new StringBuilder();
        
        sb.AppendLine("你是一位专业的文档信息提取助手。");
        sb.AppendLine();
        sb.AppendLine("【任务】");
        sb.AppendLine("从以下文档中提取指定字段和表格信息。");
        sb.AppendLine();
        
        // 字段部分
        if (fields.Any())
        {
            sb.AppendLine("【需要提取的字段】");
            foreach (var field in fields.OrderBy(f => f.SortOrder))
            {
                var required = field.IsRequired ? "（必须）" : "";
                sb.AppendLine($"{field.FieldCode}（{field.DataType}）{required}: {field.FieldName}");
            }
            sb.AppendLine();
        }
        
        // 表格部分
        if (tables.Any())
        {
            sb.AppendLine("【需要提取的表格】");
            foreach (var table in tables.OrderBy(t => t.SortOrder))
            {
                sb.AppendLine($"表格: {table.TableCode}");
                sb.AppendLine($"  名称: {table.TableName}");
                if (!string.IsNullOrEmpty(table.Description))
                    sb.AppendLine($"  描述: {table.Description}");
                
                foreach (var field in table.Fields.OrderBy(f => f.SortOrder))
                {
                    var required = field.IsRequired ? "（必须）" : "";
                    sb.AppendLine($"  - {field.FieldCode}（{field.DataType}）{required}: {field.FieldName}");
                }
                sb.AppendLine();
            }
        }
        
        sb.AppendLine("【输出格式】");
        sb.AppendLine("返回JSON格式:");
        sb.AppendLine("{");
        
        // 字段输出示例
        foreach (var field in fields)
        {
            var example = GetExampleValue(field.DataType);
            sb.AppendLine($"  \"{field.FieldCode}\": {example}, // {field.FieldName}");
        }
        
        // 表格输出示例
        foreach (var table in tables)
        {
            sb.AppendLine($"  \"{table.TableCode}\": [");
            sb.AppendLine("    {");
            foreach (var field in table.Fields)
            {
                var example = GetExampleValue(field.DataType);
                sb.AppendLine($"      \"{field.FieldCode}\": {example}, // {field.FieldName}");
            }
            sb.AppendLine("    }");
            sb.AppendLine("  ],");
        }
        
        sb.AppendLine("}");
        sb.AppendLine();
        sb.AppendLine("【文档内容】");
        sb.AppendLine("{document_content}");
        
        return sb.ToString();
    }
    
    private string GetExampleValue(string dataType)
    {
        return dataType.ToLower() switch
        {
            "string" => "\"示例文本\"",
            "number" => "123.45",
            "date" => "\"2024-01-15\"",
            "boolean" => "true",
            "money" => "1000000.00",
            "percent" => "0.35",
            _ => "\"\""
        };
    }
}
```

```
1. 只输出JSON，不要其他解释
2. fieldCode使用英文驼峰命名
3. 对于明显的关键信息（如企业名称、日期、金额）识别为字段
4. 对于列表类信息（如多个股东、多个产品）识别为表格";

        // 3. 调用千问API
        var config = _configService.GetConfig();
        var response = await CallQwenApiAsync(config, analyzePrompt);
        
        // 4. 解析结果
        return ParseAnalysisResult(response);
    }
    
    private async Task<string> ExtractContentAsync(string filePath, string docType)
    {
        return docType.ToLower() switch
        {
            "word" => await ExtractWordContent(filePath),
            "excel" => await ExtractExcelContent(filePath),
            "txt" => await File.ReadAllTextAsync(filePath),
            _ => throw new NotSupportedException($"不支持的文档类型: {docType}")
        };
    }
    
    private async Task<string> ExtractWordContent(string filePath)
    {
        // 使用NPOI提取Word全部文字
        using var doc = new XWPFDocument(File.OpenRead(filePath));
        var sb = new StringBuilder();
        
        foreach (var para in doc.Paragraphs)
        {
            sb.AppendLine(para.Text);
        }
        
        // 提取表格内容
        foreach (var table in doc.Tables)
        {
            sb.AppendLine("[表格开始]");
            foreach (var row in table.Rows)
            {
                var cells = row.GetTableCells().Select(c => c.GetText()).ToList();
                sb.AppendLine(string.Join(" | ", cells));
            }
            sb.AppendLine("[表格结束]");
        }
        
        return sb.ToString();
    }
    
    private async Task<string> ExtractExcelContent(string filePath)
    {
        // 使用NPOI提取Excel内容
        using var fs = File.OpenRead(filePath);
        IWorkbook workbook = new XSSFWorkbook(fs);
        var sb = new StringBuilder();
        
        for (int i = 0; i < workbook.NumberOfSheets; i++)
        {
            var sheet = workbook.GetSheetAt(i);
            sb.AppendLine($"[工作表: {sheet.SheetName}]");
            
            for (int row = 0; row <= sheet.LastRowNum; row++)
            {
                var excelRow = sheet.GetRow(row);
                if (excelRow == null) continue;
                
                var cells = new List<string>();
                for (int col = 0; col < excelRow.LastCellNum; col++)
                {
                    cells.Add(excelRow.GetCell(col)?.ToString() ?? "");
                }
                sb.AppendLine(string.Join(" | ", cells));
            }
            sb.AppendLine();
        }
        
        return sb.ToString();
    }
}
```

## 八、文档预览技术方案

### 8.1 需求分析

系统需要支持以下文档格式的在线预览：

| 格式类型 | 具体格式 | 使用场景 |
|---------|---------|---------|
| 图片 | JPG, PNG, GIF, BMP | 证件扫描件、照片 |
| 文档 | DOCX, PDF | 营业执照、合同、报告 |
| 表格 | XLSX, CSV | 财务报表、数据统计 |
| 文本 | TXT | 说明文件、日志 |

### 8.2 技术方案选型

#### 方案对比

| 方案 | 支持格式 | 实现方式 | 优点 | 缺点 |
|-----|---------|---------|------|------|
| **vue-office** ⭐推荐 | docx/xlsx/pdf | 纯前端 Vue 组件 | 一站式、样式还原好、Vue2/3兼容 | 不支持旧版 .doc |
| **kkFileView** | 几乎所有格式 | 后端转换+前端预览 | 格式最全、兼容性好 | 需部署后端服务 |
| **pdfjs-dist** | pdf | 纯前端 | PDF 官方方案 | 仅支持 PDF |
| **SheetJS** | xlsx/csv | 纯前端 | Excel 解析成熟 | 样式需自行处理 |

#### 推荐方案：vue-office + 原生标签（纯前端）

**理由**:
1. 项目已使用 Vue 3，vue-office 完美兼容
2. 无需额外部署后端服务
3. 样式还原度高，满足认证审核需求
4. 社区活跃，文档完善

### 8.3 组件封装设计

```vue
<!-- components/DocPreview/UniversalPreview.vue -->
<template>
  <div class="universal-preview">
    <!-- 图片预览 -->
    <template v-if="isImage">
      <el-image
        :src="src"
        :preview-src-list="[src]"
        fit="contain"
        style="width: 100%; height: 100%"
      />
    </template>
    
    <!-- Word 文档 -->
    <template v-else-if="isDocx">
      <vue-office-docx
        :src="src"
        style="height: 100%"
        @rendered="onRendered"
        @error="onError"
      />
    </template>
    
    <!-- Excel 表格 -->
    <template v-else-if="isExcel">
      <vue-office-excel
        :src="src"
        style="height: 100%"
        @rendered="onRendered"
        @error="onError"
      />
    </template>
    
    <!-- PDF 文档 -->
    <template v-else-if="isPdf">
      <vue-office-pdf
        :src="src"
        style="height: 100%"
        @rendered="onRendered"
        @error="onError"
      />
    </template>
    
    <!-- 文本文件 -->
    <template v-else-if="isText">
      <pre class="text-preview">{{ textContent }}</pre>
    </template>
    
    <!-- 不支持的格式 -->
    <template v-else>
      <div class="unsupported-tip">
        <el-icon :size="48"><Document /></el-icon>
        <p>该文件格式暂不支持预览</p>
        <el-button type="primary" @click="downloadFile">下载查看</el-button>
      </div>
    </template>
  </div>

<script setup>
import { computed, ref, watch } from 'vue';
import VueOfficeDocx from '@vue-office/docx';
import VueOfficeExcel from '@vue-office/excel';
import VueOfficePdf from '@vue-office/pdf';
import '@vue-office/docx/lib/index.css';
import '@vue-office/excel/lib/index.css';
import '@vue-office/pdf/lib/index.css';

const props = defineProps({
  src: String,        // 文件地址或 ArrayBuffer
  fileType: String,   // 文件类型: jpg/png/docx/xlsx/pdf/txt
  fileName: String    // 文件名（用于判断类型）
});

const emit = defineEmits(['rendered', 'error']);

const textContent = ref('');

// 根据文件名或类型判断
const ext = computed(() => {
  if (props.fileType) return props.fileType.toLowerCase();
  if (props.fileName) {
    return props.fileName.split('.').pop().toLowerCase();
  }
  return '';
});

const isImage = computed(() => ['jpg', 'jpeg', 'png', 'gif', 'bmp', 'webp'].includes(ext.value));
const isDocx = computed(() => ext.value === 'docx');
const isExcel = computed(() => ['xlsx', 'xls', 'csv'].includes(ext.value));
const isPdf = computed(() => ext.value === 'pdf');
const isText = computed(() => ['txt', 'md', 'json', 'xml'].includes(ext.value));

// 加载文本内容
watch(() => props.src, async (newSrc) => {
  if (isText.value && typeof newSrc === 'string') {
    try {
      const response = await fetch(newSrc);
      textContent.value = await response.text();
    } catch (e) {
      textContent.value = '加载失败: ' + e.message;
    }
  }
}, { immediate: true });

const onRendered = () => emit('rendered');
const onError = (e) => emit('error', e);
const downloadFile = () => window.open(props.src, '_blank');
</script>
```

### 8.6 后端支持（文件访问）

后端需要提供文件预览接口：

```csharp
/// <summary>
/// 获取文件预览地址（支持 MinIO 文件）
/// </summary>
[HttpGet("api/file/preview/{fileCode}")]
public async Task<IActionResult> GetFilePreview(string fileCode)
{
    var file = await _fileService.GetAsync(fileCode);
    if (file == null) return NotFound();
    
    // 从 MinIO 获取临时访问 URL（1小时有效）
    var previewUrl = await _minioService.GetPresignedUrlAsync(
        file.StoragePath, 
        expirySeconds: 3600
    );
    
    // 如果是文本文件，直接返回内容
    if (IsTextFile(file.FileName))
    {
        var content = await _minioService.GetFileContentAsync(file.StoragePath);
        return Content(content, "text/plain");
    }
    
    // 其他文件返回重定向到 MinIO 直链
    return Redirect(previewUrl);
}
```

### 8.7 注意事项

1. **旧版 .doc 格式**：vue-office 不支持，建议引导用户上传 .docx 格式，或考虑 kkFileView 方案
2. **大文件处理**：超过 10MB 的文件建议先下载再预览，避免浏览器卡顿
3. **样式还原**：vue-office 样式还原度约 90%，复杂表格可能略有差异
4. **安全性**：文件预览 URL 应设置有效期，防止未授权访问
