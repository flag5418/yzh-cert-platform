```csharp
public async Task<List<string>> WarmUpAsync(WfExecutionTask task, WorkflowConfig config)
{
    _logger.CacheWarmupStart(task.Code, config.Nodes.Count);
    var sw = Stopwatch.StartNew();
    var cacheKeys = new List<string>();

    foreach (var node in config.Nodes)
    {
        if (node.NodeType == "docField")
        {
            var docType = node.Config["docType"].ToString();
            var ruleCode = node.Config["ruleCode"].ToString();
            var fieldCode = node.Config["fieldCode"].ToString();

            if (docType == "standard")
            {
                var value = await _docExtractionService.TestFieldAsync(ruleCode, fieldCode, "standard");
                var cacheKey = $"wf:task:{task.Code}:doc:field:{ruleCode}:{fieldCode}";
                await _cache.Set(cacheKey, JsonSerializer.Serialize(value), 3600);
                cacheKeys.Add(cacheKey);
                _logger.CacheSet(cacheKey, JsonSerializer.Serialize(value).Length);
            }
            else if (docType == "enterprise")
            {
                var value = await _docExtractionService.TestFieldAsync(ruleCode, fieldCode, "enterprise");
                var cacheKey = $"wf:task:{task.Code}:ent:field:{task.EnterpriseCode}:{ruleCode}:{fieldCode}";
                await _cache.Set(cacheKey, JsonSerializer.Serialize(value), 3600);
                cacheKeys.Add(cacheKey);
                _logger.CacheSet(cacheKey, JsonSerializer.Serialize(value).Length);
            }
        }
        // ... docTable、skill 类似处理
    }

    sw.Stop();
    _logger.CacheWarmupDone(task.Code, cacheKeys.Count, (int)sw.ElapsedMilliseconds);
    return cacheKeys;
}
```

```csharp
public async Task<T> GetAsync<T>(string taskCode, string cacheKeySuffix)
{
    var cacheKey = $"wf:task:{taskCode}:{cacheKeySuffix}";
    var cached = await _cache.Get(cacheKey);
    
    if (cached != null)
    {
        _logger.CacheHit(cacheKey);
        return JsonSerializer.Deserialize<T>(cached);
    }
    
    _logger.CacheMiss(cacheKey);
    return default;
}
```

```csharp
private Dictionary<string, object> ResolveInputs(
    Dictionary<string, string> inputBindings,
    Dictionary<string, object> upstreamOutputs)
{
    var result = new Dictionary<string, object>();
    foreach (var kv in inputBindings)
    {
        if (kv.Value.StartsWith("constant:"))
        {
            result[kv.Key] = kv.Value.Substring("constant:".Length);
        }
        else
        {
            result[kv.Key] = upstreamOutputs.GetValueOrDefault(kv.Value);
        }
    }
    return result;
}
```

**执行流程**：
```csharp
public async Task<ItemExecutionResult> RunItemAsync(
    WfExecutionTask task,
    WfExecutionTaskItem item,
    WorkflowConfig config)
{
    // 1. 解析拓扑
    var parsed = _parser.Parse(config);
    _logger.PathEnum(task.Code, item.Code, parsed.Paths.Count);
    
    // 2. 跨路径共享输出
    var sharedOutputs = new Dictionary<string, object>();
    var pathResults = new List<PathResult>();
    
    // 3. 逐路径执行（路径内节点串行阻塞）
    for (int i = 0; i < parsed.Paths.Count; i++)
    {
        _logger.PathStart(task.Code, item.Code, i, parsed.Paths[i].Count);
        var result = await ExecutePathAsync(parsed.Paths[i], task, item, sharedOutputs, i);
        pathResults.Add(result);
        
        if (result.Status == "completed")
            _logger.PathDone(task.Code, item.Code, i, "completed");
        else
            _logger.PathFail(task.Code, item.Code, i, result.FailedAtNodeId, result.Error);
    }
    
    // 4. 聚合 NC 结果
    return AggregateResults(pathResults);
}

private async Task<PathResult> ExecutePathAsync(
    List<WorkflowNodeConfig> path,
    WfExecutionTask task,
    WfExecutionTaskItem item,
    Dictionary<string, object> sharedOutputs,
    int pathIndex)
{
    foreach (var node in path)
    {
        // 检查跨路径复用
        var existing = await QueryNodeExecution(task.Code, item.Code, node.NodeId);
        if (existing != null && existing.ExecStatus == "completed")
        {
            sharedOutputs[node.NodeId] = DeserializeOutput(existing.OutputJson);
            _logger.NodeReuse(task.Code, item.Code, node.NodeId);
            continue;
        }
        
        _logger.NodeStart(task.Code, item.Code, node.NodeId, node.NodeType, node.Title);
        
        // 解析输入
        var upstreamOutputs = ResolveInputs(node.Inputs, sharedOutputs);
        _logger.NodeInputs(task.Code, item.Code, node.NodeId, upstreamOutputs);
        
        // 执行节点
        var sw = Stopwatch.StartNew();
        var result = await _nodeExecutor.ExecuteAsync(
            node, task.Code, item.Code, task.EnterpriseCode, upstreamOutputs);
        sw.Stop();
        
        // 写入 wf_node_execution
        await SaveNodeExecution(task.Code, item.Code, node, result);
        
        if (!result.Success)
        {
            _logger.NodeFail(task.Code, item.Code, node.NodeId, result.Error, (int)sw.ElapsedMilliseconds);
            return new PathResult
            {
                Status = "failed",
                FailedAtNodeId = node.NodeId,
                Error = result.Error
            };
        }
        
        _logger.NodeOutput(task.Code, item.Code, node.NodeId, result.Output, (int)sw.ElapsedMilliseconds);
        _logger.NodeDone(task.Code, item.Code, node.NodeId, (int)sw.ElapsedMilliseconds);
        
        sharedOutputs[node.NodeId] = result.Output;
    }
    
    return new PathResult
    {
        Status = "completed",
        Output = sharedOutputs
    };
}
```

**验证点**：
- 简单路径（start → end）：能正确执行
- 分支路径（start → docField → branch → 2个end）：能枚举 2 条路径，分别执行
- 跨路径复用：branch 分叉后的公共节点不重复执行
- 节点失败：当前路径终止，其他路径继续
- 日志输出：完整记录执行过程

```csharp
    // 4. 预热缓存
    var cacheKeys = await _cacheWarmer.WarmUpAsync(task, config);
    task.CacheKeys = JsonSerializer.Serialize(cacheKeys);
    await _taskService.UpdateTaskAsync(task);
    
    // 5. 同步执行
    _logger.TaskStart(task.Code, 1);
    var result = await _interpreter.RunItemAsync(task, item, config);
    
    // 6. 更新 task 状态
    sw.Stop();
    task.IsSuccess = result.Success;
    task.ResultSummary = JsonSerializer.Serialize(result);
    task.ItemStatus = "completed";
    task.DurationMs = (int)sw.ElapsedMilliseconds;
    await _taskService.UpdateTaskAsync(task);
    
    _logger.TaskDone(task.Code, "completed", (int)sw.ElapsedMilliseconds);
    
    // 7. 清理缓存
    await _cacheCleaner.CleanUpAsync(task.Code, cacheKeys);
    
    return JsonNormal(new { success = true, data = result });
}
```

**每个 Phase 完成后可独立验证**：
- Phase 1 完成：能解析 rule_json，枚举路径
- Phase 2 完成：能独立执行单个节点，缓存命中
- Phase 3 完成：能端到端执行完整工作流
- Phase 4 完成：能通过 API 调用，数据库正确写入
- Phase 5 完成：能在前端页面验证工作流

## 五、风险与应对

| 风险 | 影响 | 应对 |
|------|------|------|
| `DocExtractionRuleService.TestFieldAsync` 依赖 AI 分析 | 配置验证可能慢 | TEST 模式下只读 sample_data，不调 AI |
| SkillExecutor 反射调用可能失败 | 节点执行异常 | 完善错误处理，记录到 wf_node_execution |
| 前端 LogicFlow 节点叠加图标困难 | 执行状态展示不美观 | 先用简单的 DOM 覆盖层，后续优化 |
| 跨路径复用的并发问题 | 多路径同时写入同一节点 | 使用数据库唯一键 + 乐观锁 |
| 缓存预热耗时过长 | 任务开始延迟 | 并行预热 + 进度日志 |
| Redis 连接不稳定 | 缓存读取失败 | 降级到直接查 DB + 日志告警 |
