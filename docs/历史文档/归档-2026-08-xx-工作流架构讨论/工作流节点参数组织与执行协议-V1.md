# 工作流节点参数组织与执行协议-V1

> 本文档描述工作流引擎中前端节点的参数组织结构，供后端执行引擎开发参考。

## 一、节点分类

```
节点
├── 特殊节点（引擎内置，不落 wf_skill 表）
│   ├── start   —— 开始节点（singleton）
│   ├── end     —— 结束节点
│   ├── branch  —— 条件分支节点
│   ├── ai_node —— AI 节点
│   ├── docField —— 文档字段节点
│   └── docTable —— 文档表格节点
│
└── 功能节点（后端 Skill 表注册，前端动态加载）
    ├── compare       —— 比较节点
    ├── get_field     —— 获取字段节点
    └── ...           —— 其他自定义节点
```

## 二、节点元数据模型（NodeMeta）

```typescript
interface NodeMeta {
  // === 基础标识 ===
  classCode: string;        // 节点分类编码（如 start/end/branch/ai_node/docField/compare）
  className: string;        // 节点显示名称（如 "开始"/"条件分支"/"AI 节点"）
  category: string;         // 分类分组（control/ai/data + 后端 Skill 的 category）
  color: string;            // 节点配色（hex 格式，如 "#67C23A"）
  icon: string;             // 节点图标（如 "VideoPlay"/"ChatDotRound"）

  // === 行为标志 ===
  singleton: boolean;       // 是否单例（仅允许存在一个，如 start）
  testable: boolean;        // 是否可测试（是否显示"测试节点"按钮）
  renameable: boolean;      // 是否可重命名（start 不可重命名）
  description: string;      // 节点说明（tooltip 显示）

  // === 端口声明 ===
  inputPorts: PortDef[];    // 输入端口列表（面板渲染 + 运行时绑定）
  outputPorts: PortDef[];   // 输出端口列表（画布锚点 + 下游引用）

  // === 面板 Schema ===
  panelSchema: PanelField[]; // 节点特有配置字段（AI 提示词/docSelect/tableSelect 等）
}
```

## 三、端口定义（PortDef）

```typescript
interface PortDef {
  name: string;             // 端口标识（如 "input"/"condition"/"result"/"fieldValue"）
  label: string;            // 端口显示名（如 "输入数据"/"条件值"/"汇聚结果"）
  type: PortType;           // 端口数据类型
  description: string;      // 端口说明
  bindMode: BindMode;       // 绑定模式（见下文）
  required: boolean;        // 是否必填
  maxIn?: number;           // 最大入边数（默认 1，end 的 result 为 999）
  display?: PortDisplay;    // 显示控制（visible/hidden，hidden 不在面板展示）
  role?: PortRole;          // 端口角色（anchor = 画布锚点，如 branch 的 success/failure）
  defaultValue?: any;       // 默认值
  enumSource?: string;      // 字典编码（bindMode=Enum 时使用）
  options?: EnumOption[];   // Enum 模式的静态选项
}

// 端口数据类型
type PortType = 'json' | 'string' | 'number' | 'boolean' | 'signal';

// 显示控制
type PortDisplay = 'visible' | 'hidden';

// 端口角色
type PortRole = 'anchor' | 'data';
```

## 四、BindMode 三分法

```typescript
type BindMode = 'Link' | 'LinkOrConstant' | 'Enum';

// ┌───────────────┬─────────────────────────────────────────────────────────────┐
// │ BindMode      │ 行为                                                     │
// ├───────────────┼─────────────────────────────────────────────────────────────┤
// │ Link          │ 仅支持画板连线。面板渲染为下拉选择器（选择画布上的节点）。   │
// │               │ 运行时值 = 上游节点的输出引用。                              │
// ├───────────────┼─────────────────────────────────────────────────────────────┤
// │ LinkOrConstant│ 支持连线 + 手动输入常量。面板渲染为下拉 + 编辑按钮切换。     │
// │               │ 运行时值 = 上游节点引用 或 常量值。                          │
// ├───────────────┼─────────────────────────────────────────────────────────────┤
// │ Enum          │ 仅字典选择。面板渲染为下拉选择器（调后台字典或静态 options）。│
// │               │ 运行时值 = 枚举值。                                        │
// └───────────────┴─────────────────────────────────────────────────────────────┘
```

## 五、面板 Schema（PanelField）

```typescript
interface PanelField {
  field: string;            // 字段路径（支持嵌套，如 "config.prompt"/"config.ruleCode"）
  label: string;            // 面板显示名（如 "提示词"/"文档"/"字段"）
  type: PanelFieldType;     // 控件类型
  required: boolean;        // 是否必填
  description: string;      // 字段说明（hint 显示）
  defaultValue?: any;       // 默认值
  options?: EnumOption[];   // select 类型的选项
}

// 控件类型
type PanelFieldType =
  | 'textarea'     // 多行文本（AI prompt）
  | 'switch'       // 开关（boolean 配置）
  | 'select'       // 下拉选择（静态选项）
  | 'doc-select'   // 文档选择（调 api/DocExtractionRule/configured-rules）
  | 'field-select' // 字段选择（调 api/DocExtractionRule/{ruleCode}/fields-tables）
  | 'table-select' // 表格选择（同上）
  | 'input';       // 单行文本
```

## 六、workflow_config JSON 结构（落库格式）

```typescript
interface WorkflowConfig {
  version: number;                          // 配置版本（当前 = 1）
  workflowType: string;                     // 工作流类型（如 "validation"）
  nodes: ConfigNode[];                      // 节点列表
  edges: ConfigEdge[];                      // 边列表
  outputConfig: OutputConfig;               // 输出配置
  glossary: string;                         // 术语表 / 说明
}

interface ConfigNode {
  nodeId: string;                           // 节点 ID（格式：{classCode}_n{序号}）
  nodeType: string;                         // 节点类型（同 classCode）
  title: string;                            // 节点显示名（用户可修改）
  skillCode?: string;                       // 功能节点编码（仅 skill 类型）
  config?: Record<string, any>;             // 节点特有配置（如 prompt/ruleCode/fieldCode）
  inputs?: Record<string, string>;          // 输入端口值（端口名 → 值/引用）
  outputs?: Record<string, any>;            // 输出端口值（运行时填充）
  inputPorts?: PortDef[];                   // 输入端口声明（可选，特殊节点有默认值）
  outputPorts?: PortDef[];                  // 输出端口声明（可选）
}

interface ConfigEdge {
  source: string;                           // 源节点 ID
  target: string;                           // 目标节点 ID
  sourceHandle?: string;                    // 源输出锚点（如 "success"/"failure"）
  targetHandle?: string;                    // 目标输入端口（如 "input"/"result"）
}

interface OutputConfig {
  [key: string]: string | OutputRef;        // 输出别名 → 引用
}

interface OutputRef {
  ref: string;                              // 引用（如 "ai_n1.content"）
  default?: any;                            // 默认值
}
```

## 七、依赖图构建规则

```
依赖图 = 显式边 ∪ 隐式引用 ∪ AI 提示词引用

1. 显式边：edges[].source → edges[].target
   - 运行时顺序依赖
   - 锚点类型：null（默认）/ success / failure

2. 隐式引用：扫描每个节点 inputs[].value
   - 精确匹配：nX.port（如 "docField_n1.fieldValue"）
   - 宽松匹配：纯 nX（如 "docTable_n1"，默认引用 result 端口）
   - 语义：该端口消费上游节点输出（运行时先获取上游值再执行本节点）

3. AI 提示词引用：扫描 ai_node.config.prompt 中的 {{nX.port}} 语法
   - 意义：提示词中嵌入了上游节点的输出值
   - 处理：编译时替换为实际值
```

### 7.1 输入端口值语义

| 值格式 | 含义 | 示例 |
|--------|------|------|
| `""` | 未绑定 | `"input": ""` |
| `"nX"` | 引用上游节点 X（默认 result 端口） | `"input": "docTable_n1"` |
| `"nX.port"` | 引用上游节点 X 的 port 端口 | `"input": "docField_n1.fieldValue"` |
| `"常量值"` | 直接常量（LinkOrConstant 模式） | `"input": "ISO9001"` |
| `"ctx.xxx"` | 运行时上下文（引擎注入） | `"condition": "ctx.org_code"` |

## 八、运行时执行协议

### 8.1 执行顺序

```
1. 拓扑排序（Kahn 算法，入度为 0 的节点优先执行）
2. branch 节点按条件选择 success/failure 分支（非两边都执行）
3. 孤立数据源节点（无入边但被引用）在执行时直接获取其配置 + 执行提取
```

### 8.2 节点执行上下文

```typescript
interface ExecutionContext {
  // 引擎注入（start 节点自动填充）
  orgCode: string;          // 企业编码
  standardCode: string;     // 标准编码（如 "ISO9001"）
  phaseCode: string;        // 阶段编码
  fileCode: string;         // 文件编码（可选）

  // 运行时状态
  nodeOutputs: Map<NodeId, NodeOutput>; // 每个节点的执行结果
  variables: Map<string, any>;          // 变量（预留）
}

interface NodeOutput {
  result?: any;             // 主输出（默认端口 = "result"）
  [portName: string]: any;  // 其他端口输出（如 ai_node 的 content/json/confidence）
}
```

#### ai_node 节点

```typescript
// 执行逻辑：渲染提示词（替换 {{nX.port}}）+ 调 LLM
async function executeAINode(ctx: ExecutionContext, config: ConfigNode): Promise<NodeOutput> {
  // 1. 解析输入
  const inputValue = resolveInput(ctx, config.inputs.input);

  // 2. 渲染提示词
  let prompt = config.config.prompt;
  prompt = renderTemplate(prompt, {
    ...ctx.nodeOutputs,
    input: inputValue  // {{input}} 引用
  });

  // 3. 调 LLM
  const llmResult = await llmService.call({
    prompt,
    jsonMode: config.config.jsonMode ?? true
  });

  return {
    content: llmResult.text,
    json: config.config.jsonMode ? llmResult.parsed : undefined,
    confidence: llmResult.confidence
  };
}
```

#### branch 节点

```typescript
// 执行逻辑：读取条件值，返回分支选择结果
function executeBranch(ctx: ExecutionContext, config: ConfigNode): { branch: 'success' | 'failure' } {
  const conditionNode = config.inputs.condition;
  const boolResult = ctx.nodeOutputs.get(conditionNode)?.result;
  return { branch: boolResult ? 'success' : 'failure' };
}
```

#### end 节点

```typescript
// 执行逻辑：汇聚上游结果，输出最终结果
function executeEnd(ctx: ExecutionContext, config: ConfigNode, incomingOutputs: NodeOutput[]): NodeOutput {
  return {
    result: incomingOutputs.map(o => o.result)
  };
}
```

### 8.4 输入引用解析

```typescript
function resolveInput(ctx: ExecutionContext, inputValue: string): any {
  if (!inputValue) return undefined;

  // ctx 引用
  if (inputValue.startsWith('ctx.')) {
    return ctx[inputValue.slice(4)];
  }

  // nX.port 精确引用
  const preciseMatch = inputValue.match(/^([a-zA-Z_]+_n\d+)\.([a-zA-Z_][a-zA-Z0-9_]*)$/);
  if (preciseMatch) {
    const [, nodeId, port] = preciseMatch;
    return ctx.nodeOutputs.get(nodeId)?.[port];
  }

  // nX 宽松引用（默认 result 端口）
  const looseMatch = inputValue.match(/^([a-zA-Z_]+_n\d+)$/);
  if (looseMatch) {
    const [, nodeId] = looseMatch;
    return ctx.nodeOutputs.get(nodeId)?.result;
  }

  // 常量
  return inputValue;
}

function renderTemplate(template: string, data: Record<string, any>): string {
  // 替换 {{nX.port}} 和 {{nX}} 和 {{input}}
  return template.replace(/\{\{([\w.]+)\}\}/g, (match, key) => {
    const value = data[key];
    return value !== undefined ? String(value) : match;
  });
}
```

## 九、模板语法

### 9.1 AI 提示词模板

```
{{nX.portName}}     —— 引用节点 nX 的 portName 输出
{{nX}}              —— 引用节点 nX 的默认（result）输出
{{input}}           —— 引用当前节点的 input 端口值
{{ctx.xxx}}         —— 运行时上下文变量
```

### 9.2 特殊约定

- 空值不渲染（保留占位符原文）
- 嵌套对象自动 `JSON.stringify`
- 字符串值原样输出

## 十、错误处理

```typescript
interface NodeError {
  nodeId: string;
  nodeType: string;
  code: 'EXECUTION_FAILED' | 'INPUT_MISSING' | 'LLM_ERROR' | 'TIMEOUT';
  message: string;
  cause?: Error;
}

// 执行策略：
// 1. 单节点失败 → 整体失败（不跳过）
```
