## 一、概述

### 1.1 背景

在标准目录管理系统中，上传的文档（Word/Excel/TXT）包含大量结构化信息（如企业名称、信用代码、日期等）。目前需要人工查看并录入，效率低下且容易出错。

### 1.2 目标

构建一个**AI驱动的文档数据提取系统**，实现：

1. **智能规则生成**: AI分析文档，自动生成提取规则
2. **精准数据提取**: 基于规则自动提取文档中的关键字段
3. **人工确认修正**: 提供友好的界面确认和修正提取结果
4. **批量处理能力**: 支持队列化的批量文档处理

### 1.3 范围（第一阶段）

| 文档类型 | 支持状态 | 提取方式 |
|---------|---------|---------|
| Word (.docx) | ✅ 支持 | NPOI + AI规则 |
| Excel (.xlsx) | ✅ 支持 | NPOI + AI规则 |
| TXT (.txt) | ✅ 支持 | 正则 + AI规则 |
| PDF | ⏳ 第二阶段 | OCR + AI规则 |
| 图片 | ⏳ 第二阶段 | OCR + AI规则 |

### 1.4 核心原则

1. **复用现有**: 充分利用现有NPOI代码，不做大规模重构
2. **内聚架构**: 提取服务内聚在主服务中，不单独部署Docker
3. **安全控制**: API Key + IP白名单保护内部接口
4. **AI解耦**: AI只生成规则，不直接参与提取执行

```csharp
namespace VOL.WebApi.Controllers.Internal
{
    /// <summary>
    /// 内部API基类 - 需要特殊认证
    /// </summary>
    [Route("api/internal/extraction")]
    [InternalApiAuth]
    public abstract class InternalExtractionControllerBase : ControllerBase
    {
        // 内部API公共逻辑
    }
    
    /// <summary>
    /// 内部API认证过滤器
    /// </summary>
    public class InternalApiAuthAttribute : Attribute, IAuthorizationFilter
    {
        public void OnAuthorization(AuthorizationFilterContext context)
        {
            // 1. IP白名单检查
            var remoteIp = context.HttpContext.Connection.RemoteIpAddress;
            if (!IsAllowedIp(remoteIp))
            {
                context.Result = new ForbidResult();
                return;
            }
            
            // 2. API Key检查
            var apiKey = context.HttpContext.Request.Headers["X-Internal-API-Key"].FirstOrDefault();
            if (string.IsNullOrEmpty(apiKey) || !ValidateApiKey(apiKey))
            {
                context.Result = new UnauthorizedObjectResult(new { message = "Invalid API Key" });
                return;
            }
        }
        
        private bool IsAllowedIp(IPAddress ip)
        {
            if (ip == null) return false;
            return ip.IsLoopback() || 
                   ip.ToString().StartsWith("192.168.") ||
                   ip.ToString().StartsWith("10.");
        }
        
        private bool ValidateApiKey(string apiKey)
        {
            var validKey = "sk-internal-doc-extraction-2026"; // 实际应从配置读取
            return apiKey == validKey;
        }
    }
}
```

```csharp
    }
```

```csharp
    /// 获取提取结果
    /// </summary>
```

```csharp
            var operation = context.Parameters.GetValue<string>("operation");
            var filePath = context.FilePath;
            
            try
            {
                object result = operation switch
                {
                    "findText" => await FindTextAsync(filePath, context.Parameters),
                    "extractParagraphs" => await ExtractParagraphsAsync(filePath),
                    "extractAfter" => await ExtractAfterAsync(filePath, context.Parameters),
                    "extractBetween" => await ExtractBetweenAsync(filePath, context.Parameters),
                    _ => throw new NotSupportedException($"Operation {operation} not supported")
                };
                
                return new SkillResult 
                { 
                    Success = true, 
                    Data = result,
                    Confidence = 0.95
                };
            }
            catch (Exception ex)
            {
                return new SkillResult 
                { 
                    Success = false, 
                    ErrorMessage = ex.Message 
                };
            }
        }
        
        private async Task<string> FindTextAsync(string filePath, Dictionary<string, object> parameters)
        {
            var keyword = parameters.GetValue<string>("keyword");
            return await _npoiService.FindParagraphContainingAsync(filePath, keyword);
        }
        
        private async Task<string> ExtractAfterAsync(string filePath, Dictionary<string, object> parameters)
        {
            var keyword = parameters.GetValue<string>("keyword");
            var separator = parameters.GetValue<string>("separator");
            var maxLength = parameters.GetValue<int>("maxLength", 100);
            
            var text = await _npoiService.FindParagraphContainingAsync(filePath, keyword);
            if (string.IsNullOrEmpty(text)) return null;
            
            var index = text.IndexOf(separator);
            if (index < 0) return null;
            
            var result = text.Substring(index + separator.Length).Trim();
            return result.Length > maxLength ? result.Substring(0, maxLength) : result;
        }
        
        private async Task<string> ExtractBetweenAsync(string filePath, Dictionary<string, object> parameters)
        {
            var startText = parameters.GetValue<string>("startText");
            var endText = parameters.GetValue<string>("endText");
            
            var fullText = await _npoiService.ExtractAllTextAsync(filePath);
            
            var startIndex = fullText.IndexOf(startText);
            if (startIndex < 0) return null;
            
            startIndex += startText.Length;
            var endIndex = fullText.IndexOf(endText, startIndex);
            if (endIndex < 0) return null;
            
            return fullText.Substring(startIndex, endIndex - startIndex).Trim();
        }
        
        private async Task<List<string>> ExtractParagraphsAsync(string filePath)
        {
            return await _npoiService.ExtractAllParagraphsAsync(filePath);
        }

            }
```

```csharp
                };
            }
```

```csharp
namespace YZH.Framework.DocumentExtraction.AI
{
    /// <summary>
    /// 规则验证器
    /// </summary>
    public class RuleValidator
    {
        private readonly SkillRegistry _skillRegistry;
        
        public async Task<RuleValidationResult> ValidateAsync(RuleDefinition rule)
        {
            var result = new RuleValidationResult { IsValid = true };
            
            foreach (var field in rule.Fields)
            {
                var fieldResult = await ValidateFieldAsync(field);
                if (!fieldResult.IsValid)
                {
                    result.IsValid = false;
                    result.Errors.Add($"字段 '{field.FieldName}': {fieldResult.ErrorMessage}");
                }
            }
            
            // 计算整体置信度
            result.Confidence = rule.Fields.Average(f => f.Confidence);
            
            return result;
        }
        
        private async Task<FieldValidationResult> ValidateFieldAsync(FieldDefinition field)
        {
            // 验证skillChain中的每个skill都存在
            foreach (var step in field.SkillChain)
            {
                var skill = _skillRegistry.Get(step.Skill);
                if (skill == null)
                {
                    return FieldValidationResult.Failed($"Skill '{step.Skill}' 不存在");
                }
                
                // 验证必需参数
                foreach (var param in skill.Parameters.Where(p => p.Required))
                {
                    if (!step.Params.ContainsKey(param.Name))
                    {
                        return FieldValidationResult.Failed($"缺少必需参数 '{param.Name}'");
                    }
                }
            }
            
            return FieldValidationResult.Success();
        }
    }
    
    public class RuleValidationResult
    {
        public bool IsValid { get; set; }
        public double Confidence { get; set; }
        public List<string> Errors { get; set; } = new();
    }
}
```
