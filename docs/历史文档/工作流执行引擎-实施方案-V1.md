private Dictionary<string, object> ResolveInputs(
    Dictionary<string, string> inputBindings,
    Dictionary<string, object> upstreamOutputs)
{
    var result = new Dictionary<string, object>();
    foreach (var kv in inputBindings)
    {
        if (kv.Value.StartsWith("constant:"))
        {
            result[kv.Key] = kv.Value.Substring("constant:".Length);
        }
        else
        {
            // kv.Value 是源节点 ID
            result[kv.Key] = upstreamOutputs.GetValueOrDefault(kv.Value);
        }
    }
    return result;
}
```

    public string Error { get; set; }

}

public class PathResult
{
    public string PathId { get; set; }
    public string Status { get; set; }      // completed / failed
    public string FailedAtNodeId { get; set; }
    public string Error { get; set; }

```

**执行流程**：
```csharp
public async Task<ItemExecutionResult> RunItemAsync(
    WfExecutionTask task,
    WfExecutionTaskItem item,
    WorkflowConfig config)
{
    // 1. 解析拓扑
    var parsed = _parser.Parse(config);
    
    // 2. 跨路径共享输出
    var sharedOutputs = new Dictionary<string, object>();
    var pathResults = new List<PathResult>();
    
    // 3. 逐路径执行（路径内节点串行阻塞）
    foreach (var path in parsed.Paths)
    {
        var result = await ExecutePathAsync(path, task, item, sharedOutputs);
        pathResults.Add(result);
    }
    
    // 4. 聚合 NC 结果
    return AggregateResults(pathResults);
}

private async Task<PathResult> ExecutePathAsync(
    List<WorkflowNodeConfig> path,
    WfExecutionTask task,
    WfExecutionTaskItem item,
    Dictionary<string, object> sharedOutputs)
{
    foreach (var node in path)
    {
        // 检查跨路径复用
        var existing = await QueryNodeExecution(task.Code, item.Code, node.NodeId);
        if (existing != null && existing.ExecStatus == "completed")
        {
            sharedOutputs[node.NodeId] = DeserializeOutput(existing.OutputJson);
            continue;
        }
        
        // 解析输入
        var upstreamOutputs = ResolveInputs(node.Inputs, sharedOutputs);
        
        // 执行节点
        var result = await _nodeExecutor.ExecuteAsync(
            node, task.Code, item.Code, task.EnterpriseCode, upstreamOutputs);
        
        // 写入 wf_node_execution
        await SaveNodeExecution(task.Code, item.Code, node, result);
        
        if (!result.Success)
        {
            return new PathResult
            {
                Status = "failed",
                FailedAtNodeId = node.NodeId,
                Error = result.Error
            };
        }
        
        sharedOutputs[node.NodeId] = result.Output;
    }
    
    return new PathResult
    {
        Status = "completed",
        Output = sharedOutputs
    };
}
```

**验证点**：
- 简单路径（start → end）：能正确执行
- 分支路径（start → docField → branch → 2个end）：能枚举 2 条路径，分别执行
- 跨路径复用：branch 分叉后的公共节点不重复执行
- 节点失败：当前路径终止，其他路径继续

public async Task<IActionResult> TestRun([FromBody] ExecuteWorkflowRequest request)

## 四、实施顺序与依赖关系

```
Phase 1（模型解析）     ← 无依赖，最先做
    ↓
Phase 2（节点执行）     ← 依赖 Phase 1 的解析模型
    ↓
Phase 3（路径驱动）     ← 依赖 Phase 1 + 2
    ↓
Phase 4（API 对接）     ← 依赖 Phase 3
    ↓
Phase 5（前端对接）     ← 依赖 Phase 4
```

**每个 Phase 完成后可独立验证**：
- Phase 1 完成：能解析 rule_json，枚举路径
- Phase 2 完成：能独立执行单个节点
- Phase 3 完成：能端到端执行一个完整工作流（通过单元测试）
- Phase 4 完成：能通过 Postman/curl 调用 API
- Phase 5 完成：能在前端页面验证工作流

## 五、风险与应对

| 风险 | 影响 | 应对 |
|------|------|------|
| `DocExtractionRuleService.TestFieldAsync` 依赖 AI 分析 | 配置验证可能慢 | TEST 模式下只读 sample_data，不调 AI |
| SkillExecutor 反射调用可能失败 | 节点执行异常 | 完善错误处理，记录到 wf_node_execution |
| 前端 LogicFlow 节点叠加图标困难 | 执行状态展示不美观 | 先用简单的 DOM 覆盖层，后续优化 |
| 跨路径复用的并发问题 | 多路径同时写入同一节点 | 使用数据库唯一键 + 乐观锁 |
