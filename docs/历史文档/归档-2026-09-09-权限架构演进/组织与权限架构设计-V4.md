# 多租户体系认证平台——组织与权限架构设计（V4）

> **版本**：V4.0 | **日期**：2026-09-08 | **状态**：修订版
>
> **修订说明**：基于 V3.0 讨论，引入**分层架构**设计——通用系统层 + 体系认证业务层。
>
> **核心设计**：
> 1. 通用系统：机构-人员-角色-菜单管理（标准能力）
> 2. 体系认证业务：按认证特点建立组织，自动调用通用接口同步权限
> 3. 两层解耦：业务层享受通用层的菜单/接口控制，同时保留独立权限管理能力

### 2.3 通用权限中间件

```csharp
// 通用权限中间件（所有系统共用）
public class CommonPermissionMiddleware
{
    public async Task<bool> CheckAsync(UserContext user, string apiPath, string httpMethod)
    {
        // 1. 检查登录状态
        if (user == null) return false;
        
        // 2. SuperAdmin 拥有所有权限
        if (user.UserType == "super_admin") return true;
        
        // 3. 检查接口权限
        var apiCode = $"{httpMethod.ToUpper()} {apiPath}";
        var hasApiPermission = await _roleService.HasApiPermissionAsync(
            user.RoleCodes, apiCode);
        if (!hasApiPermission) return false;
        
        // 4. 检查菜单权限（前端路由守卫使用）
        var hasMenuPermission = await _roleService.HasMenuPermissionAsync(
            user.RoleCodes, apiPath);
        if (!hasMenuPermission) return false;
        
        return true;
    }
}
```

### 3.3 体系认证组织创建流程（自动调用通用接口）

```csharp
// 体系认证业务服务：创建认证机构时自动同步通用系统
public class CertOrgService
{
    private readonly IOrgService _orgService;  // 通用机构服务
    private readonly IUserService _userService; // 通用用户服务
    private readonly IRoleService _roleService; // 通用角色服务
    private readonly IMenuService _menuService; // 通用菜单服务
    
    // 创建认证机构（含机构管理员）
    public async Task CreateCertOrgAsync(CertOrgCreateRequest request)
    {
        // 1. 创建机构（通用接口）
        var deptCode = Guid.NewGuid().ToString("N");
        await _orgService.AddAsync(new OrgDepartment
        {
            Code = deptCode,
            Name = request.CbName,
            ParentCode = "sys-root",
            DeptType = "cert_org"
        });
        
        // 2. 创建机构管理员用户（通用接口）
        var adminUserCode = Guid.NewGuid().ToString("N");
        var adminUserId = await _userService.AddAsync(new SysUser
        {
            Code = adminUserCode,
            UserName = request.AdminUserName,
            Password = HashPassword(request.AdminPassword),
            RealName = request.AdminRealName,
            DeptCode = deptCode,
            Status = "active"
        });
        
        // 3. 分配角色（通用接口）
        await _roleService.AssignRoleAsync(adminUserCode, "cb_admin");
        
        // 4. 关联机构扩展信息（业务层）
        await _certOrgAttachService.AddAsync(new CertOrgAttach
        {
            DeptCode = deptCode,
            CbName = request.CbName,
            CbCode = request.CbCode,
            CreditCode = request.CreditCode,
            Status = "active"
        });
        
        return adminUserCode;
    }
    
    // 创建虚拟机构（含审核组长）
    public async Task CreateVirtualOrgAsync(VirtualOrgCreateRequest request)
    {
        // 1. 创建虚拟机构（通用接口）
        var deptCode = Guid.NewGuid().ToString("N");
        await _orgService.AddAsync(new OrgDepartment
        {
            Code = deptCode,
            Name = request.OrgName,
            ParentCode = request.ParentDeptCode,
            DeptType = "virtual_org"
        });
        
        // 2. 创建审核组长用户（通用接口）
        var leadUserCode = Guid.NewGuid().ToString("N");
        var leadUserId = await _userService.AddAsync(new SysUser
        {
            Code = leadUserCode,
            UserName = request.LeadUserName,
            Password = HashPassword(request.LeadPassword),
            RealName = request.LeadRealName,
            DeptCode = deptCode,
            Status = "active"
        });
        
        // 3. 分配角色（通用接口）
        await _roleService.AssignRoleAsync(leadUserCode, "team_lead");
        
        // 4. 关联虚拟机构扩展信息（业务层）
        await _virtualOrgAttachService.AddAsync(new VirtualOrgAttach
        {
            DeptCode = deptCode,
            CbCode = request.CbCode,
            OrgType = "audit_team",
            LeadUserCode = leadUserCode,
            DataScope = "self"
        });
        
        return leadUserCode;
    }
    
    // 添加审核员
    public async Task AddAuditorAsync(AuditorAddRequest request)
    {
        // 1. 创建审核小组（通用接口）
        var deptCode = Guid.NewGuid().ToString("N");
        await _orgService.AddAsync(new OrgDepartment
        {
            Code = deptCode,
            Name = request.TeamName,
            ParentCode = request.VirtualOrgCode,
            DeptType = "audit_team"
        });
        
        // 2. 创建审核员用户（通用接口）
        var userCode = Guid.NewGuid().ToString("N");
        var userId = await _userService.AddAsync(new SysUser
        {
            Code = userCode,
            UserName = request.UserName,
            Password = HashPassword(request.Password),
            RealName = request.RealName,
            Phone = request.Phone,
            Email = request.Email,
            DeptCode = deptCode,
            Status = "active"
        });
        
        // 3. 分配角色（通用接口）
        await _roleService.AssignRoleAsync(userCode, "auditor");
        
        // 4. 关联审核员扩展信息（业务层）
        await _auditorAttachService.AddAsync(new AuditorAttach
        {
            DeptCode = deptCode,
            AuditorNo = request.AuditorNo,
            Specialties = request.Specialties,
            AnnualQuota = request.AnnualQuota,
            Status = "active"
        });
        
        return userCode;
    }
    
    // 添加企业用户
    public async Task AddEnterpriseUserAsync(EnterpriseUserAddRequest request)
    {
        // 1. 创建企业用户组（通用接口）
        var deptCode = Guid.NewGuid().ToString("N");
        await _orgService.AddAsync(new OrgDepartment
        {
            Code = deptCode,
            Name = request.UserName,
            ParentCode = request.VirtualOrgCode,
            DeptType = "enterprise_group"
        });
        
        // 2. 创建企业用户（通用接口）
        var userCode = Guid.NewGuid().ToString("N");
        var userId = await _userService.AddAsync(new SysUser
        {
            Code = userCode,
            UserName = request.UserName,
            Password = HashPassword(request.Password),
            RealName = request.RealName,
            Phone = request.Phone,
            Email = request.Email,
            DeptCode = deptCode,
            Status = "active"
        });
        
        // 3. 分配角色（通用接口）
        await _roleService.AssignRoleAsync(userCode, "enterprise_user");
        
        // 4. 关联企业用户扩展信息（业务层）
        await _enterpriseUserAttachService.AddAsync(new EnterpriseUserAttach
        {
            DeptCode = deptCode,
            EnterpriseCode = request.EnterpriseCode,
            DataScope = "enterprise_only",
            Status = "active"
        });
        
        return userCode;
    }
}
```

### 5.2 数据权限自动注入

```csharp
// 后端 Base Controller 自动注入数据权限过滤
public abstract class YzhControllerBase<V> : ControllerBase
{
    protected virtual List<FilterItem> OnBuildingQuery(RequestCondition[]? conditions)
    {
        var user = _userContext.GetCurrent();
        var filters = new List<FilterItem>(conditions ?? Array.Empty<RequestCondition>());
        
        // 根据用户机构类型自动注入数据权限
        var attachData = _attachService.GetByUserDept(user.DeptCode);
        
        if (attachData?.DeptType == "cert_org")
        {
            // CB管理员：只能查看本CB及子机构数据
            var deptCodes = _orgService.GetAllChildCodes(user.DeptCode);
            filters.Add(new FilterItem 
            { 
                Field = "dept_code", 
                Value = string.Join(",", deptCodes), 
                Operator = "in" 
            });
        }
        else if (attachData?.DeptType == "virtual_org")
        {
            // 虚拟机构：只能查看本机构数据
            filters.Add(new FilterItem 
            { 
                Field = "dept_code", 
                Value = user.DeptCode, 
                Operator = "eq" 
            });
        }
        else if (attachData?.DeptType == "audit_team")
        {
            // 审核员：只能查看被分配任务
            filters.Add(new FilterItem 
            { 
                Field = "auditor_code", 
                Value = user.Code, 
                Operator = "eq" 
            });
        }
        else if (attachData?.DeptType == "enterprise_group")
        {
            // 企业用户：只能查看本企业
            filters.Add(new FilterItem 
            { 
                Field = "enterprise_code", 
                Value = attachData.EnterpriseCode, 
                Operator = "eq" 
            });
        }
        
        return filters;
    }
}
```

## 七、关键设计优势

### 7.1 分层解耦

```
通用系统层：
  ✓ 标准 RBAC 实现
  ✓ 所有系统可复用
  ✓ 独立维护和升级

体系认证业务层：
  ✓ 按认证特点扩展
  ✓ 自动调用通用接口
  ✓ 保留独立权限管理能力
```

### 7.2 权限清晰

```
Level 1（通用）：
  - 菜单权限：sys_role_menu
  - 接口权限：sys_role_api
  - 角色分配：sys_user_role

Level 2（业务）：
  - 数据权限：attach 视图自动注入
  - 配置权限：config_scope JSON
  - 业务权限：控制器层实现
```

### 7.3 扩展简单

```
新增业务场景 = 新增 attach 表 + 新增业务角色 + 配置权限

示例：未来需要新增"外部专家"角色
  1. 新增 org_external_expert_attach 表
  2. 在 sys_role 新增 external_expert 角色
  3. 配置 menu 和 api 权限
  4. 完成！
```
