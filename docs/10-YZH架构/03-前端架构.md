## 二、CrudPageLogic<V> — 单表 Logic 基类

### 2.1 类定义

```typescript
// 文件：cert/cert-share/src/logic/CrudPageLogic.ts

export abstract class CrudPageLogic<V extends Record<string, any>> {
    // ──── 必须设置 ────
    abstract controllerName: string    // 控制器名称（如 'User'、'Role'）

    // ──── 后端配置 ────
    config = ref<EntityConfigDto | null>(null)

    // ──── 表格状态 ────
    rows = ref<V[]>([])
    loading = ref(false)
    selectedRows = ref<V[]>([])

    // ──── 分页状态 ────
    pagination = reactive({ page: 1, pageSize: 20, total: 0 })

    // ──── 搜索过滤状态 ────
    searchParams = reactive<Record<string, any>>({})
    sortField = ref<string | undefined>()
    sortOrder = ref<'asc' | 'desc' | undefined>()

    // ──── 弹窗状态 ────
    dialogVisible = ref(false)
    dialogMode = ref<'add' | 'edit'>('add')
    submitting = ref(false)
    formData = reactive<Record<string, any>>({}) as Partial<V>
}
```

### 2.2 从 config 派生的 UI 结构（computed）

```typescript
// 表格列配置（XsFlag=true 的列）
get columns(): YzhTableColumn<V>[]

// 表单字段配置（BcFlag=true 的字段）
get formFields(): YzhFormField[]

// 搜索栏字段配置（优先 SearchFields，否则从 Columns 推导）
get searchFields(): SearchField[]

// 工具栏按钮配置（从 ToolbarConfig 派生）
get toolbarButtons(): Array<{key: string, text: string, type: string, action?: string}>

// 行操作按钮配置（从 RowButtonConfig 派生）
get rowButtons(): Array<{key: string, text: string, type: string, action?: string}>

// 主键字段名（统一使用 'code'）
get primaryKey(): string { return 'code' }
```

### 2.3 核心方法

```typescript
// ──── 初始化 ────
async init(): Promise<void>           // loadConfig → loadPage
protected async loadConfig(): Promise<void>  // GET /api/{controller}/config

// ──── 数据查询 ────
async loadPage(): Promise<void>       // POST /api/{controller}/filter

// ──── 写入操作 ────
protected async add(entity: Partial<V>): Promise<V>        // POST /add
protected async update(entity: Partial<V>): Promise<V>     // POST /update
protected async delete(codes: string[]): Promise<void>     // POST /delete
async executeAction(methodName: string, row: V): Promise<void>  // POST /action/{name}

// ──── 导出导入 ────
async exportData(format?: 'excel' | 'csv', fields?: string[]): Promise<void>
async importData(file: File): Promise<any>
async downloadImportTemplate(): Promise<void>

// ──── Split 增量更新（不重新请求）───
protected removeRowByCode(code: string): void    // 删除行
protected replaceRowByCode(code: string, row: V): void  // 替换行
protected insertRow(row: V, position?: 'top' | 'bottom'): void  // 插入行
```

### 2.4 钩子方法（子类可覆盖）

```typescript
// 数据加载后
protected onDataLoaded(_rows: V[]) {}

// 新增/修改前（可修改待提交实体）
protected onBeforeAdd(_entity: Partial<V>) {}
protected onBeforeUpdate(_entity: Partial<V>) {}

// 新增/修改后
protected onAfterAdd(_entity: Partial<V>) {}
protected onAfterUpdate(_entity: Partial<V>) {}

// 删除前（返回 false 取消删除）
protected onDelete(_codes: string[]): boolean | Promise<boolean> { return true }

// 删除后
protected onAfterDelete(_codes: string[]) {}

// 准备新增数据（设置默认值）
protected onPrepareAdd(_entity: Partial<V>) {}
```

### 2.5 弹窗操作

```typescript
// 打开新增弹窗
openAddDialog()    // 使用 config.NewEntity 初始化表单

// 打开编辑弹窗
openEditDialog(row: V)  // 用 row 数据覆盖默认值

// 取消弹窗
async cancelDialog()

// 提交表单
async submitForm()
//   → add() 或 update() → Split 方法增量更新 → ElMessage.success
```

## 三、TreeTableLogic<V> — 左树右表 Logic 基类

### 3.1 继承关系

```
CrudPageLogic<V>
    │
    ▼
TreeTableLogic<V>     ← 继承单表全部能力 + 叠加树能力
```

### 3.2 新增状态

```typescript
export abstract class TreeTableLogic<V extends Record<string, any>>
    extends CrudPageLogic<V> {

    // ──── 树状态 ────
    treeData = ref<TreeNode[]>([])
    treeLoading = ref(false)
    selectedNode = ref<TreeNode | null>(null)
    treeTableConfig = ref<any>(null)  // 完整 TreeTableConfigDto

    // ──── 树节点弹窗状态 ────
    treeDialogVisible = ref(false)
    treeDialogMode = ref<'add' | 'edit'>('add')
    treeSubmitting = ref(false)
    treeFormData = reactive<Record<string, any>>({})
    treeParentNode = ref<TreeNode | null>(null)
    treeEditingNode = ref<TreeNode | null>(null)
}
```

### 3.3 快捷访问器


### 3.4 核心方法


### 3.6 完整示例（Organization）

```typescript
// cert/cert-admin/src/pages/system/organization/logic.ts
import { TreeTableLogic } from '@share/logic'
import { ref } from 'vue'

export class OrgPageLogic extends TreeTableLogic<any> {
    controllerName = 'Organization'

    // ShowDisabled 开关
    showDisabled = ref(false)

    // 表格列扩展（添加操作列）
    get columnsWithActions() {
        const cols = this.columns as any[]
        const enhanced = cols.map(c => c.prop === 'Enable' ? { ...c, slot: true } : c)
        return [...enhanced, { prop: 'actions', label: '操作', slot: true, width: 240, fixed: 'right' }]
    }

    // 表单字段扩展（添加隐藏字段）
    get formFieldsWithHidden() {
        const fields = [...this.formFields]
        if (this.dialogMode.value === 'add') {
            fields.push({ prop: 'OrgCode', label: '所属机构', type: 'text', disabled: true, span: 24 })
        }
        return fields
    }

    // 覆盖 buildFilters：添加 ShowDisabled
    protected buildFilters(extra?: Record<string, any>) {
        const filters = super.buildFilters(extra)
        if (this.showDisabled.value) {
            filters.push({ Field: 'ShowDisabled', Value: 'true', Operator: 'eq' })
        }
        return filters
    }

    // 切换 ShowDisabled
    toggleShowDisabled() {
        this.showDisabled.value = !this.showDisabled.value
        this.pagination.page = 1
        this.loadPage()
    }
}
```

## 七、页面开发模式

### 7.1 模式一：配置驱动（推荐）

适用于：标准 CRUD 页面，字段来自后端 JSON 配置。

```
┌──────────────────────────────────────────────────────────────┐
│  后端 Controller 继承 YzhControllerBase<V>                    │
│  → 提供 /config /filter /add /update /delete /action/*        │
│                                                              │
│  前端 Logic 继承 CrudPageLogic<V>                             │
│  → 自动从 /config 获取 columns / formFields / searchFields    │
│  → 提供 loadPage() / add() / update() / delete()              │
│                                                              │
│  前端 .vue 只需：                                              │
│  1. new Logic() 并 await logic.init()                         │
│  2. <YzhTable :columns="logic.columns" ... />                 │
│  3. <YzhForm :fields="logic.formFields" ... />                │
└──────────────────────────────────────────────────────────────┘
```

### 7.2 模式二：手写 API（当前已有页面）

适用于：后端尚未完成迁移、或特殊业务页面。

```
┌──────────────────────────────────────────────────────────────┐
│  前端直接调用 cert/cert-share/src/api/xxx.ts 中的手写 API 函数   │
│  → 自行管理 columns / searchFields / 弹窗状态                   │
│  → 不依赖 Logic 基类                                           │
│                                                              │
│  示例：foundation/iso-standard/index.vue                       │
│  → columns 硬编码                                              │
│  → 直接调用 getISOStandardPage(params)                         │
└──────────────────────────────────────────────────────────────┘
```

> **趋势**：新页面优先使用模式一（配置驱动），旧页面逐步迁移。
