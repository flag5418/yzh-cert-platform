## 一、权限体系表结构（待建设）

当前 `PermissionFilter.CheckPermission()` 是空实现，核心原因是**权限相关表结构未建立**。

### 1.1 需要新建的表

```sql
-- ============================================================
-- 1. 接口表（已存在，需确认字段完整性）
-- ============================================================
CREATE TABLE IF NOT EXISTS sys_api (
    id          BIGINT PRIMARY KEY AUTO_INCREMENT,
    code        VARCHAR(100) NOT NULL UNIQUE COMMENT '接口编码',
    method      VARCHAR(10) NOT NULL COMMENT 'HTTP方法: GET/POST/PUT/DELETE',
    path        VARCHAR(200) NOT NULL COMMENT '接口路径',
    module      VARCHAR(50) COMMENT '所属模块',
    name        VARCHAR(100) NOT NULL COMMENT '接口名称',
    enable      TINYINT(1) DEFAULT 1 COMMENT '是否启用',
    create_date DATETIME DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_api_module (module)
) COMMENT '接口表';

-- ============================================================
-- 2. 角色-接口关联表（已存在，需确认索引）
-- ============================================================
CREATE TABLE IF NOT EXISTS sys_role_api (
    id          BIGINT PRIMARY KEY AUTO_INCREMENT,
    role_code   VARCHAR(50) NOT NULL COMMENT '角色编码',
    api_code    VARCHAR(100) NOT NULL COMMENT '接口编码',
    create_date DATETIME DEFAULT CURRENT_TIMESTAMP,
    UNIQUE INDEX uk_role_api (role_code, api_code),
    INDEX idx_api_code (api_code)
) COMMENT '角色-接口关联表';

-- ============================================================
-- 3. 用户-角色关联表（已存在，需确认索引）
-- ============================================================
CREATE TABLE IF NOT EXISTS sys_user_role (
    id          BIGINT PRIMARY KEY AUTO_INCREMENT,
    user_code   VARCHAR(36) NOT NULL COMMENT '用户编码',
    role_code   VARCHAR(50) NOT NULL COMMENT '角色编码',
    role_id     INT NOT NULL COMMENT '角色ID',
    create_date DATETIME DEFAULT CURRENT_TIMESTAMP,
    UNIQUE INDEX uk_urole (user_code, role_code)
) COMMENT '用户-角色关联表';

-- ============================================================
-- 4. 权限缓存表（新建，用于性能优化）
-- ============================================================
CREATE TABLE IF NOT EXISTS sys_user_permission (
    id          BIGINT PRIMARY KEY AUTO_INCREMENT,
    user_code   VARCHAR(36) NOT NULL COMMENT '用户编码',
    api_code    VARCHAR(100) NOT NULL COMMENT '接口编码',
    create_date DATETIME DEFAULT CURRENT_TIMESTAMP,
    update_date DATETIME ON UPDATE CURRENT_TIMESTAMP,
    UNIQUE INDEX uk_user_api (user_code, api_code),
    INDEX idx_user_code (user_code)
) COMMENT '用户权限缓存表（角色-接口关联的展开表）';

-- ============================================================
-- 5. 数据权限规则表（新建，用于行级数据权限）
-- ============================================================
CREATE TABLE IF NOT EXISTS sys_data_scope (
    id              BIGINT PRIMARY KEY AUTO_INCREMENT,
    role_code       VARCHAR(50) NOT NULL COMMENT '角色编码',
    table_name      VARCHAR(100) NOT NULL COMMENT '数据表名',
    scope_type      VARCHAR(20) NOT NULL COMMENT '权限类型: All/Department/DepartmentAndSub/Self/Custom',
    scope_sql       TEXT COMMENT '自定义SQL条件（scope_type=Custom时使用）',
    enable          TINYINT(1) DEFAULT 1,
    create_date     DATETIME DEFAULT CURRENT_TIMESTAMP,
    update_date     DATETIME ON UPDATE CURRENT_TIMESTAMP,
    UNIQUE INDEX uk_role_table (role_code, table_name)
) COMMENT '数据权限规则表';
```

### 1.2 权限校验逻辑设计

```csharp
// 伪代码：PermissionFilter.CheckPermission() 实现思路
private bool CheckPermission(IUserContext userContext, string permissionCode)
{
    // 1. 超级管理员直接放行
    if (userContext.RoleId == RoleIds.SuperAdmin) return true;

    // 2. 从缓存获取用户权限列表（首次查询后缓存5分钟）
    var cacheKey = $"user_perm:{userContext.UserCode}";
    var userPermissions = _cacheManager.GetOrAdd(
        cacheKey,
        () => _permissionRepo.GetUserPermissions(userContext.UserCode),
        TimeSpan.FromMinutes(5)
    );

    // 3. 检查是否包含所需权限
    return userPermissions.Contains(permissionCode);
}

// 权限缓存刷新时机：
// - 用户角色变更时（OnAfterAdd/OnAfterUpdate 钩子）
// - 接口权限变更时（sys_role_api 表变更时）
// - Token 续租时（强制刷新缓存）
```

### 1.3 数据权限注入设计

```csharp
// PermissionFilter.OnAuthorization() 中增加数据权限注入
public void OnAuthorization(AuthorizationFilterContext context)
{
    // ... 原有功能权限校验 ...

    // 注入数据权限到 HttpContext
    var dataScope = _dataScopeService.GetDataScope(
        userContext.UserCode,
        userContext.RoleId,
        context.ActionDescriptor.DisplayName
    );
    context.HttpContext.Items["DataScope"] = dataScope;
}

// EntityService 中应用数据权限
public async Task<Result<PagedResult<V>>> GetPageAsync(PagerOptions options)
{
    // 从 HttpContext 获取数据权限
    var dataScope = _httpContextAccessor.HttpContext?.Items["DataScope"] as DataScope;
    if (dataScope != null)
    {
        // 根据 scope_type 注入 SQL 条件
        options.Filters = ApplyDataScopeFilter(options.Filters, dataScope);
    }
    // ... 原有分页逻辑 ...
}
```
