## 一、继承体系

```
ControllerBase (ASP.NET Core)
    │
    ▼
YzhControllerBase<V>          ← 单表 CRUD
    │  泛型约束：V : class, new()
    │  提供：/config /filter /add /update /delete /export /import /action/{name}
    │  钩子：OnBeforeAdd / OnAfterAdd / OnBeforeUpdate / OnAfterUpdate
    │         OnBeforeDelete / OnAfterDelete / OnAfterCommitted
    │         OnQueried / OnBuildingFilter / OnConfigLoading
    │
    ▼
TreeTableControllerBase<T,V>  ← 左树右表
    │  泛型约束：T : class, ITreeEntity, new() | V : class, new()
    │  继承单表全部能力
    │  新增：/tree/root /tree/children /tree/add /tree/update /tree/delete /tree/action/{name}
    │  树钩子：OnBeforeAddTree / OnAfterAddTree / OnBeforeUpdateTree
    │         / OnAfterUpdateTree / OnBeforeDeleteTree / OnAfterDeleteTree
    │  必须配置：TreeConfig（NameField/CodeField/ParentCodeField/RelateField）
    └── 典型实现：OrganizationController(Sys_Organization, Sys_User)
```

## 二、YzhControllerBase<V> — 单表 CRUD 基类

### 2.1 类定义

```csharp
// 文件：src/yzh-core/YZH.Core.Api/Controllers/YzhControllerBase.cs
// 命名空间：YZH.Core.Api.Controllers

[ApiController]
[Route("api/[controller]")]
[YZHAuthorize]   // 默认强认证，测试接口用 [YZHAnonymous]
public abstract class YzhControllerBase<V> : ControllerBase where V : class, new()
{
    protected EntityService<V> Entity { get; }
    protected EntityConfig Config { get; }
    protected IUserContext UserContext { get; }
    protected string ControllerName => GetType().Name.Replace("Controller", "");
}
```

### 2.2 构造函数

```csharp
protected YzhControllerBase(EntityService<V> entityService, IUserContext userContext)
{
    Entity = entityService;
    UserContext = userContext;
    Config = LoadConfig();   // 自动从 JSON 加载配置
}

protected virtual EntityConfig LoadConfig()
{
    if (typeof(BaseEntity).IsAssignableFrom(typeof(V)))
        return EntityConfigHelper.GetEntityConfig<V>();
    return new EntityConfig { Title = typeof(V).Name, Columns = new List<DefineColumn>() };
}
```

### 2.4 生命周期钩子（子类可覆盖）

#### 查询钩子

```csharp
// 构建过滤条件（子类可添加全局过滤，如租户隔离、ShowDisabled）
protected virtual List<FilterItem> OnBuildingFilter(List<FilterItem> filters)
    => filters;

// 查询后处理（字典翻译、脱敏）
protected virtual void OnQueried(PagedResult<V> result) { }

// 配置加载后钩子（注入动态配置）
protected virtual void OnConfigLoading(EntityConfig config) { }
```

**典型用法 — ShowDisabled 开关**（见 `OrganizationController.OnBuildingFilter`）：

```csharp
protected override List<FilterItem> OnBuildingFilter(List<FilterItem> filters)
{
    var showDisabled = false;
    var sdFilter = filters.FirstOrDefault(f => f.Field == "ShowDisabled");
    if (sdFilter != null && bool.TryParse(sdFilter.Value?.ToString(), out var sd))
    {
        showDisabled = sd;
        filters.Remove(sdFilter);
    }
    if (!showDisabled)
    {
        filters.RemoveAll(f => f.Field == "Enable");
        filters.Add(new FilterItem { Field = "Enable", Operator = "eq", Value = "1" });
    }
    return filters;
}
```

#### 新增钩子

```csharp
// 新增前（返回 false 可取消操作）
protected virtual Task<(bool ok, string? msg)> OnBeforeAdd(V entity)
    => Task.FromResult<(bool, string?)>((true, null));

// 新增后（事务内）
protected virtual Task OnAfterAdd(V entity) => Task.CompletedTask;
```

**典型用法 — 密码加密**（见 `UserController.OnBeforeAdd`）：

```csharp
protected override async Task<(bool ok, string? msg)> OnBeforeAdd(Sys_User entity)
{
    if (await Entity.ExistsByCodeAsync(entity.UserName))
        return (false, $"账号 {entity.UserName} 已存在");
    entity.UserPwd = _passwordHelper.AesEncrypt(entity.UserPwd ?? "123456");
    entity.Enable = 1;
    return (true, null);
}
```

#### 修改钩子

```csharp
protected virtual Task<(bool ok, string? msg)> OnBeforeUpdate(V entity)
    => Task.FromResult<(bool, string?)>((true, null));

protected virtual Task OnAfterUpdate(V entity) => Task.CompletedTask;
```

#### 删除钩子

```csharp
protected virtual Task<(bool ok, string? msg)> OnBeforeDelete(string[] codes)
    => Task.FromResult<(bool, string?)>((true, null));

protected virtual Task OnAfterDelete(int count) => Task.CompletedTask;
```

**典型用法 — 禁止删除含子节点的父级**（见 `OrganizationController.OnBeforeDeleteTree`）：

```csharp
protected override async Task<(bool ok, string? msg)> OnBeforeDeleteTree(Sys_Organization entity)
{
    var childCount = await TreeEntity.CountAsync(o => o.ParentCode == entity.Code);
    if (childCount.Data > 0)
        return (false, $"该机构下有 {childCount.Data} 个子机构，请先删除子机构");
    return (true, null);
}
```

#### 提交后钩子

```csharp
// 所有写入成功后调用（事务外，用于发送通知、清理缓存）
protected virtual Task OnAfterCommitted() => Task.CompletedTask;
```

### 2.5 行操作注册

```csharp
// 注册行操作（子类构造函数中调用）
protected void RegisterRowAction(string actionName, RowActionHandler handler)
    where RowActionHandler = delegate Task<Result<ApiResponse<object?>>> (V entity);

// 统一入口：POST /api/{controller}/action/{methodName}
// 前端只传 entity.Code，避免全量实体校验失败
[HttpPost("action/{methodName}")]
public virtual async Task<ActionResult<ApiResponse<object?>>> ExecuteAction(
    string methodName, [FromBody] JsonElement entityData)
{
    // 仅提取 Code 属性，构造最小实体
    var code = entityData.TryGetProperty("Code", out var cp) ? cp.GetString() : null;
    var entity = new V();
    typeof(V).GetProperty("Code")?.SetValue(entity, code);
    var result = await _rowActions[methodName.ToLower()](entity);
    return RequestResultToApiResponse(result);
}
```

### 3.4 树→表格联动

```csharp
// FilterCore 覆盖：自动注入树过滤条件
public override async Task<Result<PagedResult<V>>> FilterCore(FilterRequest request)
{
    var treeFilter = request.Filters?.FirstOrDefault(f => f.Field == TreeConfig.RelateField);

    if (treeFilter != null)
    {
        // 选中节点 → 注入过滤条件
        var otherFilters = request.Filters?.Where(f => f.Field != TreeConfig.RelateField).ToList() ?? new();
        if (!otherFilters.Any(f => f.Field == TreeConfig.RelateField))
            otherFilters.Add(new FilterItem { Field = TreeConfig.RelateField, Operator = "eq", Value = treeFilter.Value });
        request.Filters = otherFilters;
    }
    else
    {
        // 未选中 → 按 NoSelectionBehavior 决定
        if (TreeConfig.NoSelectionBehavior == "empty")
            return Result<PagedResult<V>>.Ok(new PagedResult<V>(Array.Empty<V>(), 0, request.Page, request.PageSize));
    }
    return await base.FilterCore(request);
}
```

### 3.5 操作注册与自动注入机制（核心）

> **设计理念**：Controller 注册操作 → 后端自动注入到 `/config` 响应 → 前端自动渲染按钮。

#### 两种注册方式

```csharp
// 1. 行操作（单表行按钮 / 树表表格行按钮共用）
protected void RegisterRowAction(string methodName, Func<V, Task<object?>> handler);

// 2. 树节点操作（仅树表）
protected void RegisterTreeAction(string methodName, Func<T, Task<object?>> handler);
```

#### 自动注入到 /config 响应

`GetConfig()` 被调用时，自动将已注册的操作注入到配置 DTO：

```csharp
private void InjectTreeActionsToConfig(TreeBehaviorConfigDto dto)

    // 将 _treeActions 注入到 dto.CustomActions
```

## 六、完整示例：OrganizationController

```csharp
// 文件：src/yzh-core/YZH.Core.Web/Controllers/System/OrganizationController.cs

[ApiController]
[Route("api/System/[controller]")]
[Route("api/[controller]")]
public class OrganizationController : TreeTableControllerBase<Sys_Organization, Sys_User>
{
    public OrganizationController(
        EntityService<Sys_Organization> treeEntityService,
        EntityService<Sys_User> tableEntityService,
        IUserContext userContext,
        PasswordHelper passwordHelper)
        : base(treeEntityService, tableEntityService, userContext)
    {
        // ★ 唯一配置点
        TreeConfig.NameField = "OrgName";
        TreeConfig.CodeField = "Code";
        TreeConfig.ParentCodeField = "ParentCode";
        TreeConfig.RelateField = "OrgCode";
        TreeConfig.MaxLevel = 10;
        TreeFormConfigName = "System/OrganizationForm";

        RegisterTreeAction("disable", DisableOrgRecursiveAsync);
        RegisterTreeAction("enable", EnableOrgAsync);
        RegisterRowAction("disable", DisableUserAsync);
        RegisterRowAction("enable", EnableUserAsync);
    }

    // ─── 配置加载（覆盖：指定表格实体的配置）───
    protected override EntityConfig LoadConfig()
        => EntityConfigHelper.GetConfig("System/User");

    // ─── 树节点钩子 ───
    protected override async Task<(bool ok, string? msg)> OnBeforeAddTree(Sys_Organization entity)
    {
        var nameExists = await TreeEntity.ExistsAsync(o =>
            o.ParentCode == entity.ParentCode && o.OrgName == entity.OrgName);
        if (nameExists.Data) return (false, $"同级机构下已存在名为【{entity.OrgName}】的机构");
        if (string.IsNullOrEmpty(entity.Code)) entity.Code = Guid.NewGuid().ToString("N");
        if (string.IsNullOrEmpty(entity.OrgType)) entity.OrgType = "Dept";
        return (true, null);
    }

    protected override async Task<(bool ok, string? msg)> OnBeforeDeleteTree(string[] codes)
    {
        foreach (var code in codes)
        {
            var childCount = await TreeEntity.CountAsync(o => o.ParentCode == code);
            if (childCount.Data > 0)
                return (false, $"该机构下有 {childCount.Data} 个子机构，请先删除子机构");
        }
        return (true, null);
    }

    // ─── 表格行钩子 ───
    protected override async Task<(bool ok, string? msg)> OnBeforeAdd(Sys_User entity)
    {
        if (string.IsNullOrEmpty(entity.OrgCode)) return (false, "请选择所属机构");
        var orgResult = await TreeEntity.GetByCode(entity.OrgCode);
        if (!orgResult.Success || orgResult.Data == null)
            return (false, "所属机构不存在");
        if (orgResult.Data.Enable == 0)
            return (false, "所属机构已禁用，无法添加人员");
        if (await Entity.ExistsAsync(u => u.UserName == entity.UserName))
            return (false, $"账号 {entity.UserName} 已存在");
        entity.UserPwd = _passwordHelper.AesEncrypt(entity.UserPwd ?? "123456");
        entity.Enable = 1;
        return (true, null);
    }

    protected override void OnQueried(PagedResult<Sys_User> result)
    {
        foreach (var item in result.Items)
        {
            if (!string.IsNullOrEmpty(item.PhoneNo) && item.PhoneNo.Length >= 7)
                item.PhoneNo = $"{item.PhoneNo[..3]}****{item.PhoneNo[^4..]}";
            item.RoleName = item.RoleId switch
            {
                1 => "超级管理员", 10 => "总管理员", 20 => "审核管理员",
                21 => "审核组长", 22 => "普通审核员", 30 => "企业账号", _ => "未知"
            };
            item.EnableDesc = item.Enable == 0 ? "已禁用" : "启用";
        }
    }

    protected override List<FilterItem> OnBuildingFilter(List<FilterItem> filters)
    {
        var showDisabled = false;
        var sdFilter = filters.FirstOrDefault(f => f.Field == "ShowDisabled");
        if (sdFilter != null && bool.TryParse(sdFilter.Value?.ToString(), out var sd))
        { showDisabled = sd; filters.Remove(sdFilter); }
        if (!showDisabled)
        {
            filters.RemoveAll(f => f.Field == "Enable");
            filters.Add(new FilterItem { Field = "Enable", Operator = "eq", Value = "1" });
        }
        return filters;
    }

    // ─── 行操作 ───
    private async Task<Result<ApiResponse<object?>>> DisableUserAsync(Sys_User entity)
    {
        var result = await Entity.GetByCode(entity.Code);
        if (!result.Success || result.Data == null)
            return Result<ApiResponse<object?>>.Fail("人员不存在");
        if (result.Data.RoleId == 1)
            return Result<ApiResponse<object?>>.Fail("不能禁用超级管理员账号");
        result.Data.Enable = 0;
        var updateResult = await Entity.Update(result.Data, UserContext.ClientIp);
        return updateResult.Success
            ? Result<ApiResponse<object?>>.Ok(ApiResponse<object?>.Ok("已禁用该人员"))
            : Result<ApiResponse<object?>>.Fail(updateResult.Error);
    }

    // ─── 树节点操作 ───
    private async Task<object?> DisableOrgRecursiveAsync(Sys_Organization entity)
    {
        var result = await TreeEntity.GetByCode(entity.Code);
        if (!result.Success || result.Data == null) return "机构不存在";
        var org = result.Data;
        org.Enable = 0;
        await TreeEntity.Update(org, UserContext.ClientIp);
        var childCount = await DisableChildrenRecursive(org.Code);
        return $"已禁用机构及其 {childCount} 个子机构/用户";
    }
}
```
