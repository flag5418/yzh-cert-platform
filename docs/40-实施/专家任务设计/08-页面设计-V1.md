```ts
const context = ref<CreatableContext | null>(null)

async function loadContext() {
  const res = await expertTaskApi.getCreatableContext({
    enterpriseCode: form.EnterpriseCode,
    stageCode: form.StageCode,
  })
  if (!res.success) { ElMessage.error(res.message); return }
```

### 4.2 分区规则（按 `ReviewStatus` + `SkipCategory`）

| 分区 | 条件 | 排序 |
|---|---|---|
| 待认可 | `ReviewStatus='pending'` | 条款号 |
| 已认可 | `ReviewStatus='acknowledged'` | 条款号 |
| 已修改 | `ReviewStatus='modified'` | 修改时间倒序 |
| 人工检查 | `SkipCategory='manual_mode'` | 条款号 |

```ts
  if (!props.enterpriseCode) return
  const res = await expertTaskApi.getActiveLock({ EnterpriseCode: props.enterpriseCode })
  lockInfo.value = res.success ? res.data : null
}

const uploadDisabled = computed(() => lockInfo.value?.locked ?? false)
const lockTip = computed(() =>
  lockInfo.value?.locked
    ? `该企业有进行中的任务【${lockInfo.value.taskNumber}】（${lockInfo.value.taskTypeName}），请先完成后再上传/替换/删除文件`
    : ''
)
```

模板：
```vue
<el-tooltip :content="lockTip" :disabled="!lockInfo?.locked">
  <span>
    <el-button type="primary" :disabled="uploadDisabled" @click="openUpload">上传文件</el-button>
  </span>
</el-tooltip>
```

> ⚠️ `el-button` 的 `disabled` 会吞掉 tooltip 事件，必须用 `<span>` 包一层（上面已处理）。

## 十、守卫合规自检（R1-R12）

| 守卫 | 检查项 | 本期自检 |
|---|---|---|
| R1 | 字段名 PascalCase 三处一致 | `ItemName` / `TaskNumber` / `ReviewStatus` 等全部 PascalCase |
| R2 | 禁止 `view-grid`/`VolBox`/`VolForm`/`VolProvider` | 全部用 `el-*` + 自研组件 |
| R3 | 禁止 `axios` | 一律 `yzhApi` |
| R4 | `.vue` 内禁 `fetch(` | 无 |
| R5 | 禁止手写 `handleAdd`/`handleSubmit` 等 | P1 用 `useSingleTable` 骨架；P2-P6 是自定义流程页，无 CRUD 骨架 |
| R6 | 禁止 `Enable` 字段 | 全部 `IsValid` |
| R7 | `EnableField` 值必须是 `'IsValid'` | 管理端 `ValidationRule.json` 同步改 |
| R8 | 组件必须在 `yzh.vue.core`/`cert-share`/端目录 | share 组件在 `cert-share`，端内组件在 `cert-auditor` |
| R9 | 路由命名与菜单一致 | 见 §1.1 |
| R10 | 三层同构 | Controller / api / pages 同名 |
| R11 | 禁止 `src/old/**` | 无引用 |
| R12 | 路由↔菜单双向一致 | P2-P5 登记为隐藏子菜单；P6 配菜单 |
