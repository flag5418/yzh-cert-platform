## 零、强制铁律（DDL 写之前必读）

| 铁律 | 要求 |
|---|---|
| **七 · 列名三处一致** | DB 列名 = C# 属性名 = TS 字段名，**PascalCase 逐字一致** |
| **八 · 字符集** | 建表**必须显式** `COLLATE=utf8mb4_general_ci`；⛔ 只写 `DEFAULT CHARSET=utf8mb4`（不带 COLLATE）会落到 `utf8mb4_0900_ai_ci`，不是库默认值 |
| **九 · Enable 零容忍** | 启用/禁用唯一字段 = `IsValid`（int，0/1）。⛔ 数据库中不允许任何表存在 `Enable`/`enable` 列 |
| **九 · 验证 SQL** | 用 `CONVERT(COLUMN_NAME USING utf8mb4) COLLATE utf8mb4_bin` 做大小写检查，否则 `NOT REGEXP '^[A-Z]'` **永远返回 0 行**（假阴性） |
| **准则 A · Id/Code** | `Id` **永不**进 WHERE / 关联 / `Id>0` 分流。定位删除更新传参**只用 `Code`** |
| **十 · SQL 外置** | 脚本放 `scripts/db/`，`.sh` 只做编排，不内嵌 SQL |

## 二、新建表 DDL

### 2.1 `tsk_task` — 任务头

```sql
-- ============================================================
-- Table: tsk_task（专家任务头）
-- 说明：NC 检查 与 报告生成 共用一张表，用 TaskType 区分（D01）
--       用 ScopeType 区分全局/局部（D01）
--       1 个任务按标准自动拆 N 个子任务（D02）
-- ============================================================
USE `yzh_cert_platform`;
SET NAMES utf8mb4 COLLATE utf8mb4_general_ci;

CREATE TABLE `tsk_task` (
  `Id`               bigint      NOT NULL AUTO_INCREMENT COMMENT '主键ID',
  `Code`             varchar(36) COLLATE utf8mb4_general_ci NOT NULL COMMENT '全局唯一编码（GUID，业务键）',
  `OrgCode`          varchar(50) COLLATE utf8mb4_general_ci DEFAULT NULL COMMENT '专家工作区编码（租户，★非认证机构）',
  `CreateBy`         varchar(50) COLLATE utf8mb4_general_ci DEFAULT NULL,
  `CreateName`       varchar(100) COLLATE utf8mb4_general_ci DEFAULT NULL COMMENT '创建人姓名（冗余，防改名后追溯断裂）',
  `CreateTime`       datetime     DEFAULT CURRENT_TIMESTAMP,
  `UpdateBy`         varchar(50) COLLATE utf8mb4_general_ci DEFAULT NULL,
  `UpdateTime`       datetime     DEFAULT NULL,
  `DeleteBy`         varchar(50) COLLATE utf8mb4_general_ci DEFAULT NULL,
  `DeleteTime`       datetime     DEFAULT NULL,
  `Status`           varchar(50) COLLATE utf8mb4_general_ci DEFAULT NULL COMMENT '业务状态（保留）',
  `Sort`             int          DEFAULT 0   COMMENT '排序号',
  `Remark`           varchar(500) COLLATE utf8mb4_general_ci DEFAULT NULL,
  `IsDeleted`        tinyint(1)   NOT NULL DEFAULT 0,
  `IsValid`          int          NOT NULL DEFAULT 1 COMMENT '有效标志（1=有效，0=无效）',

  `TaskNumber`       varchar(50) COLLATE utf8mb4_general_ci NOT NULL COMMENT '任务编号 TSK-yyyyMMdd-NNNN（仅展示，不作关联键）',
  `TaskType`         varchar(20) COLLATE utf8mb4_general_ci NOT NULL COMMENT '任务类型：NC_CHECK | REPORT_GENERATE',
  `ScopeType`        varchar(20) COLLATE utf8mb4_general_ci NOT NULL COMMENT '范围类型：FULL=全局 | PARTIAL=局部',
  `ScopeSnapshot`    json         DEFAULT NULL COMMENT '局部任务勾选项快照（Code 数组），FULL 时为 NULL',

  `EnterpriseCode`   varchar(36) COLLATE utf8mb4_general_ci NOT NULL COMMENT '企业编码（cert_enterprise.Code）',
  `EnterpriseName`   varchar(200) COLLATE utf8mb4_general_ci DEFAULT NULL COMMENT '企业名称（冗余快照）',

  `StageCode`        varchar(36) COLLATE utf8mb4_general_ci NOT NULL COMMENT '阶段编码（★cert_cert_stage.Code，不是 StageCode 业务码）',
  `StageName`        varchar(100) COLLATE utf8mb4_general_ci DEFAULT NULL COMMENT '阶段名称（冗余快照）',

  `StandardCodes`    json         DEFAULT NULL COMMENT '涉及标准 Code 数组（冗余，列表页免 JOIN）',
  `StandardNames`    json         DEFAULT NULL COMMENT '涉及标准名称数组',
  `StandardCount`    int          DEFAULT 0   COMMENT '涉及标准数',

  `TaskStatus`       varchar(20) COLLATE utf8mb4_general_ci NOT NULL DEFAULT 'draft'
                     COMMENT '任务状态：draft | gap_detected | pending_run | running | pending_review | approved | cancelled | failed',
  `TotalItemCount`   int          DEFAULT 0   COMMENT '任务项总数',
  `AckedCount`       int          DEFAULT 0   COMMENT '已认可数（acknowledged）',
  `ModifiedCount`    int          DEFAULT 0   COMMENT '已修改数（modified）',
  `SkippedCount`     int          DEFAULT 0   COMMENT '已跳过数（skipped）',
  `PendingCount`     int          DEFAULT 0   COMMENT '待认可数（pending）★批准前必须为 0',
  `FailedCount`      int          DEFAULT 0   COMMENT '执行失败数',
  `GapCount`         int          DEFAULT 0   COMMENT '未处理数据缺口数（★submit 前必须为 0）',
  `Progress`         decimal(5,2) DEFAULT 0.00 COMMENT '执行进度 %（队列执行进度，非认可进度）',

  `SkipAllGaps`      tinyint(1)   NOT NULL DEFAULT 0 COMMENT '专家是否点了「跳过全部补录」',
  `SkipAllGapsTime`  datetime     DEFAULT NULL,

  `SubmitTime`       datetime     DEFAULT NULL COMMENT '提交执行时间',
  `FinishTime`       datetime     DEFAULT NULL COMMENT '全部队列结束时间',
  `ApproveBy`        varchar(50) COLLATE utf8mb4_general_ci DEFAULT NULL,
  `ApproveName`      varchar(100) COLLATE utf8mb4_general_ci DEFAULT NULL,
  `ApproveTime`      datetime     DEFAULT NULL,
  `ApproveRemark`    varchar(500) COLLATE utf8mb4_general_ci DEFAULT NULL,

  PRIMARY KEY (`Id`),
  UNIQUE KEY `uk_code`          (`Code`),
  UNIQUE KEY `uk_org_tasknum`   (`OrgCode`, `TaskNumber`),
  KEY `idx_bizlock`             (`OrgCode`, `EnterpriseCode`, `StageCode`, `TaskStatus`, `IsValid`),
  KEY `idx_ent_list`            (`OrgCode`, `EnterpriseCode`, `CreateTime`),
  KEY `idx_status_list`         (`OrgCode`, `TaskStatus`, `CreateTime`),
  KEY `idx_type_list`           (`OrgCode`, `TaskType`, `CreateTime`),
  KEY `idx_stage_code`          (`StageCode`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci
  COMMENT='专家任务（NC检查/报告生成）';
```

> **`idx_bizlock` 是业务锁专用索引**（02 号 §6.2）。锁检查在每次文件操作时触发，**必须走索引不能全表扫**。字段顺序按等值列在前、范围列在后设计。

```sql
  `Conformity`     varchar(20) COLLATE utf8mb4_general_ci DEFAULT NULL

  `Severity`       varchar(20) COLLATE utf8mb4_general_ci DEFAULT NULL

  `ContentText`    longtext    COLLATE utf8mb4_general_ci
                   COMMENT '★结果正文：NC=不符合描述；报告=章节正文（纯文本，D19）',
  `EvidenceRef`    text        COLLATE utf8mb4_general_ci COMMENT '★客观证据引用（NC 专用）',

  -- ─── 认可状态（D12 强制：所有结果必须经专家认可）───

  `IsModified`     tinyint(1)   NOT NULL DEFAULT 0 COMMENT '是否被专家修改过（导出「结论来源」列用）',
  `ReviewBy`       varchar(50) COLLATE utf8mb4_general_ci DEFAULT NULL,
  `ReviewName`     varchar(100) COLLATE utf8mb4_general_ci DEFAULT NULL,
  `ReviewTime`     datetime     DEFAULT NULL,

  `NcCode`         varchar(36) COLLATE utf8mb4_general_ci DEFAULT NULL COMMENT '生成的 NC 记录编码（audit_nonconformity.Code）',
  `NcCount`        int          DEFAULT 0 COMMENT '本项产生的 NC 数（当前 0/1，保留扩展）',

  KEY `idx_itemcode`         (`ItemCode`),

  KEY `idx_enterprise`       (`OrgCode`, `EnterpriseCode`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci
  COMMENT='专家任务项（NC检查项 + 报告章节 统一表）';
```

> **`uk_task_item` 唯一键的作用**：防止 D20「不支持追加」被绕过 —— 即使有人手工调接口追加已有项，也会被唯一键拦住。

### 2.4 `tsk_task_queue` — 队列头

```sql
-- ============================================================
-- Table: tsk_task_queue（专家任务队列头）
-- 说明：新建专用队列，不复用 yzh_queue（D10）
--   理由：① yzh_queue 的 GetQueueListAsync 无租户过滤（跨租户泄露 H12）
--         ② yzh_queue.ScopeKey 双口径（file_convert=ConfigCode / doc_extract=EnterpriseCode）
--         ③ 文档转换(IO密集) 与 NC/报告(LLM密集) 调度需求不同
--   粒度：1 标准子任务 = 1 队列（1:1）
--   ScopeKey 口径：★ 唯一 = SubTaskCode（不复用双口径）
-- ============================================================
CREATE TABLE `tsk_task_queue` (
  `Id`            bigint      NOT NULL AUTO_INCREMENT COMMENT '主键ID',
  `Code`          varchar(36) COLLATE utf8mb4_general_ci NOT NULL COMMENT '全局唯一编码（GUID，业务键）',
```

> ```sql
> `GapKey` varchar(300) GENERATED ALWAYS AS (
>     CONCAT(GapType,'|', IFNULL(FieldCode,''),'|', IFNULL(TableCode,''),'|', SourceItemCode)
> ) STORED,
> UNIQUE KEY `uk_gap` (`TaskCode`, `GapKey`)
> ```

### 4.1 R2 · 阶段口径修复（阻塞项）

**问题**：`cert_validation_rule.PhaseCode` 与 `cert_report_template.PhaseCode` 的 FK 指向 `cert_phase_definition`（**0 行的孤儿表**，前端零引用、控制器已删路由），而 `cert_enterprise_stage.StageCode` 关联的是 `cert_cert_stage.Code`。两个口径不一致 → 规则配不出来、查不出来。

```sql
-- scripts/db/fix/fix-task-stage-code-align-2026-09-29.sql
USE `yzh_cert_platform`;
SET NAMES utf8mb4 COLLATE utf8mb4_general_ci;

-- ────────────────────────────────────────────────────────────
-- 1. 数据迁移：PhaseCode 从 cert_phase_definition.Code 迁到 cert_cert_stage.Code
--    cert_phase_definition 是 0 行孤儿表，理论上无数据需迁；
--    保留本段以防有脏数据，迁移后校验
-- ────────────────────────────────────────────────────────────
-- 1.1 查有脏数据吗（PhaseDefinitionCode → PhaseCode 业务码 → cert_cert_stage.Code）
SELECT vr.Code, vr.RuleCode, vr.PhaseCode AS OldPhaseCode,
       pd.PhaseCode AS PdBusinessCode, cs.Code AS NewPhaseCode, cs.StageName
FROM cert_validation_rule vr
LEFT JOIN cert_phase_definition pd ON pd.Code = vr.PhaseCode
LEFT JOIN cert_cert_stage cs       ON cs.StageCode = pd.PhaseCode
WHERE vr.PhaseCode IS NOT NULL AND cs.Code IS NULL AND vr.IsDeleted = 0;
-- 期望 0 行。若有行 → 人工确认映射关系后再处理

-- 1.2 迁移
UPDATE cert_validation_rule vr
INNER JOIN cert_phase_definition pd ON pd.Code = vr.PhaseCode
INNER JOIN cert_cert_stage cs       ON cs.StageCode = pd.PhaseCode
SET vr.PhaseCode = cs.Code, vr.UpdateBy = 'fix_task_stage_align_20260929', vr.UpdateTime = NOW();

UPDATE cert_report_template t
INNER JOIN cert_phase_definition pd ON pd.Code = t.PhaseCode
INNER JOIN cert_cert_stage cs       ON cs.StageCode = pd.PhaseCode
SET t.PhaseCode = cs.Code, t.UpdateBy = 'fix_task_stage_align_20260929', t.UpdateTime = NOW();

-- ────────────────────────────────────────────────────────────
-- 2. 改 FK 指向
-- ────────────────────────────────────────────────────────────
ALTER TABLE cert_validation_rule  DROP FOREIGN KEY fk_vrule_phase;
ALTER TABLE cert_validation_rule  ADD  CONSTRAINT fk_vrule_phase
```

### 4.2 R3 · 铁律九修复（`IsActive` → `IsValid`）

```sql
-- scripts/db/fix/fix-task-isvalid-align-2026-09-29.sql
USE `yzh_cert_platform`;
SET NAMES utf8mb4 COLLATE utf8mb4_general_ci;

-- ────────────────────────────────────────────────────────────
-- 1. cert_validation_rule：IsActive(tinyint) → IsValid(int)
--    数据迁移：0→1, 1→0（★反向，IsActive=1 表示启用，IsValid=1 才表示启用，值相同）
--    实际 IsActive=1 → IsValid=1，IsActive=0 → IsValid=0，值不变，只是类型与命名
-- ────────────────────────────────────────────────────────────
ALTER TABLE cert_validation_rule
  ADD COLUMN `IsValid` int NOT NULL DEFAULT 1 COMMENT '有效标志（1=有效，0=无效）' AFTER `IsActive`;

UPDATE cert_validation_rule SET IsValid = CASE WHEN IsActive = 1 THEN 1 ELSE 0 END;

ALTER TABLE cert_validation_rule
  DROP COLUMN `IsActive`,
  ADD KEY `idx_valid` (`OrgCode`, `StandardCode`, `PhaseCode`, `IsValid`);

-- ────────────────────────────────────────────────────────────
-- 2. rpt_report_section：新增 IsValid（该表当前只有 IsActive，且 enable 已被 DROP）
-- ────────────────────────────────────────────────────────────
ALTER TABLE rpt_report_section

ALTER TABLE rpt_report_section
  DROP COLUMN `IsActive`,
  ADD KEY `idx_template_valid` (`ReportCode`, `IsValid`, `SortOrder`);

-- ────────────────────────────────────────────────────────────
-- 3. cert_report_template：已有 IsValid，确认无需改
-- ────────────────────────────────────────────────────────────
```
