      </el-select>

        <el-option label="已失败" value="failed" />

      </el-select>
      <el-input v-model="filters.keyword" placeholder="企业名/任务号" clearable style="width: 240px">
        <template #prefix><el-icon><Search /></el-icon></template>
      </el-input>
    </div>

    <!-- 任务列表 -->
    <div v-loading="loading" class="task-list">
      <TaskCard
        v-for="task in filteredTasks"
        :key="task.code"
        :task="task"
        @click="openTaskDetail(task.code)"
        @toggle-focus="handleToggleFocus(task.code)"
      />
      <el-empty v-if="filteredTasks.length === 0 && !loading" description="暂无关注任务" />
    </div>
  </div>

<script setup>
import { ref, computed, onMounted } from 'vue'
import { useRouter } from 'vue-router'
import { useAuditTaskStore } from '@/yzh/store/auditor/useAuditTaskStore'
import CertPageHeader from '@/certcore/components/CertPageHeader.vue'
import StatCard from '../components/StatCard.vue'
import TaskCard from '../components/TaskCard.vue'

const router = useRouter()
const taskStore = useAuditTaskStore()

const filters = ref({
  standardCode: '',
  phaseCode: '',
  status: '',
  keyword: ''
})

const loading = computed(() => taskStore.loading)
const focusCount = computed(() => taskStore.focusList.length)
const activeCount = computed(() => taskStore.activeCount)
const pendingReviewCount = computed(() => taskStore.pendingReviewCount)

const filteredTasks = computed(() => {
  let list = [...taskStore.focusList]
  if (filters.value.standardCode) {
    list = list.filter(t => t.standardCode === filters.value.standardCode)
  }
  if (filters.value.status) {
    list = list.filter(t => t.status === filters.value.status)
  }
  if (filters.value.keyword) {
    const kw = filters.value.keyword.toLowerCase()
    list = list.filter(t => 
      t.enterpriseName.toLowerCase().includes(kw) ||
      t.taskNumber.toLowerCase().includes(kw)
    )
  }
  return list
})

onMounted(() => {
  taskStore.fetchFocusList()
})

function openTaskDetail(code) {
  router.push(`/Auditor/Task/Detail/${code}`)
}

function handleToggleFocus(code) {
  taskStore.unfocus(code)
}
</script>

import TaskStatusBadge from './TaskStatusBadge.vue'

  task: { type: Object, required: true }
})
defineEmits(['click', 'toggle-focus'])

const standardTagType = computed(() => {
  const map = { 'ISO9001': '', 'ISO14001': 'success', 'ISO45001': 'warning', 'ISO13485': 'danger' }
  return map[task.standardCode] || ''
})

function formatTime(time) {
  if (!time) return ''
  const diff = Date.now() - new Date(time).getTime()
  if (diff < 3600000) return `${Math.floor(diff / 60000)}分钟前`
  if (diff < 86400000) return `${Math.floor(diff / 3600000)}小时前`
  return `${Math.floor(diff / 86400000)}天前`
}
</script>

<template>
