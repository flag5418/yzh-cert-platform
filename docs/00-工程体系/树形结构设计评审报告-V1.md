<template>

      </span>
    </template>

import { ref, watch } from 'vue'
import type { TreeNode } from '@share/types/tree'

interface Props {
  data: TreeNode[]
  config?: {
    showCheckbox?: boolean
    highlightCurrent?: boolean
    defaultExpandAll?: boolean
    lazy?: boolean
    loadData?: (node: TreeNode, resolve: (children: TreeNode[]) => void) => void
  }
}

const props = defineProps<Props>()
const emit = defineEmits<{
  'node-click': [node: TreeNode]
  'check-change': [nodes: TreeNode[]]
  'node-expand': [node: TreeNode]
}>()

const treeRef = ref()

// 懒加载节点
const loadNode = async (node: any, resolve: any) => {
  if (node.level === 0) return resolve([])
  
  const treeNode = node.data as TreeNode
  if (treeNode.lazyLoadConfig?.enabled && !treeNode.lazyLoadConfig.loaded) {
    treeNode.isLoading = true
    try {
      // 调用后端接口
      const result = await yzhApi.post('/getChildren', {
        parentCode: treeNode.code,
        parentNodeType: treeNode.nodeType,
        level: treeNode.level + 1,
        childNodeType: treeNode.lazyLoadConfig.childNodeType
      })
      
      const children = result.data.items.map((item: any) => ({
        ...item,
        children: []
      }))
      treeNode.loaded = true
      treeNode.isLoading = false
      resolve(children)
    } catch (error) {
      treeNode.isLoading = false
      resolve([])
    }
  } else {
    resolve(treeNode.children || [])
  }
}

const handleNodeClick = (data: TreeNode) => {
  emit('node-click', data)
}

const handleCheckChange = (data: TreeNode, checked: boolean) => {
  emit('check-change', [data])
}

const handleNodeExpand = (data: TreeNode) => {
  emit('node-expand', data)
}
</script>

import { describe, it, expect } from 'vitest'
import { TreeUtils } from '../utils/treeUtils'

    parentCodeField: 'parentCode',
    rootParentCode: null
  }
  
  const items = [
    { code: 'A', name: '节点A', parentCode: null },
    { code: 'A1', name: '节点A1', parentCode: 'A' },
    { code: 'A2', name: '节点A2', parentCode: 'A' },
    { code: 'A11', name: '节点A11', parentCode: 'A1' }
  ]
  
  describe('buildTree', () => {
    it('应正确构建二叉树', () => {
      const tree = TreeUtils.buildTree(items, config)
      expect(tree).toHaveLength(1)
      expect(tree[0].code).toBe('A')
      expect(tree[0].children).toHaveLength(2)
    })
    
    it('应正确构建多级树', () => {
      const tree = TreeUtils.buildTree(items, config)
      const a1 = tree[0].children.find(c => c.code === 'A1')
      expect(a1?.children).toHaveLength(1)
      expect(a1?.children[0].code).toBe('A11')
    })
  })
  
  describe('getDescendants', () => {
    it('应返回所有子孙节点', () => {
      const tree = TreeUtils.buildTree(items, config)
      const descendants = TreeUtils.getDescendants(tree[0])
      expect(descendants).toHaveLength(2)
      expect(descendants.map(d => d.code)).toEqual(['A1', 'A11'])
    })
  })
  
  describe('validate', () => {
    it('应检测循环引用', () => {
      const invalidItems = [
        { code: 'A', parentCode: 'B' },
        { code: 'B', parentCode: 'A' }
      ]
      const result = TreeUtils.validate(invalidItems as any)
      expect(result.valid).toBe(false)
    })
  })
