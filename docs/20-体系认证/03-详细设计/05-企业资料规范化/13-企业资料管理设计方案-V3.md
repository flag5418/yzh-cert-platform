## 一、表结构设计

### 1.1 变更原则

| 现有表 | 新增列 | 区分方式 | 说明 |
|--------|--------|----------|------|
| `cert_standard_directory_config` | `EnterpriseCode varchar(36) NOT NULL` | `= 虚拟企业Code`（YZH-STD-ENT）= 机构模板；**真实企业 Code** = 该企业该标准该阶段的目录配置 | 同一张表，一行 = 一套完整目录；**禁 NULL**（P13：uk 对 NULL 不生效，双口径必出脏数据） |
| `cert_standard_directory_file` | `EnterpriseCode varchar(36) NOT NULL` | `= 虚拟企业Code` = 模板文件定义；**真实企业 Code** = 该企业实际上传的文件 | 模板行 + 每份企业上传各一行；**禁 NULL**（同上） |
| `cert_standard_directory_folder` | 不改 | — | 文件夹结构是共享的，不需要企业维度 |

### 1.2 `cert_standard_directory_config` 新增列

```sql

    AFTER OrgCode,

-- V3.1 变更：uk 加 OrgCode 成四列（P16，与 PathBuilder 决策⑳修订/标准目录主表 uk 对称）；列禁 NULL（P13）
-- ⚠️ 存量迁移（随 06 册 E-10 出独立脚本）：
--   ① UPDATE ... SET EnterpriseCode='YZH-STD-ENT' WHERE EnterpriseCode='';（模板行回填）
--   ② EnterpriseCode DEFAULT '' 保留（合规于禁 NULL），空串行须先清洗再改 NOT NULL
```

### 1.3 `cert_standard_directory_file` 新增列

```sql
ALTER TABLE cert_standard_directory_file
  ADD COLUMN EnterpriseCode varchar(36) NOT NULL DEFAULT '' COMMENT '企业Code：虚拟企业Code=模板定义；真实值=该企业实际上传'
    AFTER ConfigCode,
  ADD COLUMN ExtractStatus varchar(20) NOT NULL DEFAULT 'none' COMMENT '提取状态：none/pending/processing/completed/failed'
    AFTER ConvertStatus,
  ADD COLUMN ExtractMessage varchar(1024) DEFAULT NULL COMMENT '提取失败原因'
    AFTER ExtractStatus,
  ADD COLUMN MaxConfidence decimal(3,2) DEFAULT NULL COMMENT '最高提取置信度'
    AFTER ExtractMessage,
  ADD KEY idx_enterprise_file (EnterpriseCode, StandardCode, StageCode),
  ADD KEY idx_extract_status (ExtractStatus);
```

### 1.4 新增：`cert_enterprise_file_version` — 历史版本

```sql
CREATE TABLE `cert_enterprise_file_version` (
  `Id` bigint NOT NULL AUTO_INCREMENT COMMENT '主键ID',
  `Code` varchar(36) NOT NULL COMMENT '业务键GUID',
  `CreateTime` datetime NOT NULL,
  `CreateBy` varchar(64) DEFAULT NULL,
  `FileCode` varchar(36) NOT NULL COMMENT '父文件Code → cert_standard_directory_file.Code（企业上传行）',
  `EnterpriseCode` varchar(36) NOT NULL COMMENT '企业Code（冗余）',
  `VersionNumber` int NOT NULL DEFAULT '1' COMMENT '版本号（从1开始递增）',
  `FileName` varchar(500) NOT NULL COMMENT '该版本的文件名',
  `StoragePath` varchar(512) NOT NULL COMMENT '归档后的MinIO路径（_archive/xxx.doc.vN，权威算法见 PathBuilder.Archive）',
  `FileSize` bigint NOT NULL DEFAULT '0' COMMENT '该版本文件大小',
  `Reason` varchar(500) DEFAULT NULL COMMENT '替换原因',
  `IsDeleted` tinyint(1) NOT NULL DEFAULT '0',
  `IsValid` int NOT NULL DEFAULT '1',
  PRIMARY KEY (`Id`),
  UNIQUE KEY `uk_code` (`Code`),
  KEY `idx_file` (`FileCode`),
  KEY `idx_version` (`FileCode`,`VersionNumber`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci COMMENT='企业文件历史版本';
```

## 二、存储规则

### 2.1 路径格式对比

| | 标准目录（模板行） | 企业资料（企业行） |
|--|--|--|
| 首段前缀 | `/standard-directory/` | `/enterprise-documents/` |
| 第二段 | `OrgCode`（机构） | `EnterpriseCode`（企业） |
| 第三段 | `StandardCode` | `StandardCode` |
| 第四段 | `StageCode` | `StageCode` |
| 第五段+ | `FolderPath/FileName` | `FolderPath/FileName` |

### 2.2 MinIO CopyObject 能力评估

| 操作 | MinIO | 阿里云OSS |
|------|-------|-----------|
| 同桶内 CopyObject（不同路径） | ✅ 元数据操作，接近即时 | ✅ 支持 |
| CopyObject（大文件） | ✅ 同桶内仍是元数据操作 | ✅ 支持 |
| RemoveObject | ✅ 支持 | ✅ 支持 |

**结论**：同桶内 CopyObject 是廉价操作，**不需要临时存储中间层**。审核员确认映射后可直接 Copy 到最终路径。

```csharp
public async Task<FileCheckResponse> FileCheckAsync(string enterpriseCode, string stageCode)
{
    // 1. 获取该企业该阶段下的所有标准（从 cert_enterprise_stage）
    var stages = await _db.GetListAsync<CertEnterpriseStage>(
        x => x.EnterpriseCode == enterpriseCode && x.StageCode == stageCode && !x.IsDeleted);

    var result = new FileCheckResponse {
        EnterpriseCode = enterpriseCode,
        StageCode = stageCode,
        CheckTime = DateTime.Now.ToString("yyyy-MM-dd HH:mm:ss")
    };

    // 获取企业名称
    var enterprise = await _db.GetOneAsync<CertEnterprise>(x => x.Code == enterpriseCode);
    result.EnterpriseName = enterprise?.Name ?? "";

    foreach (var stage in stages)
    {
        // 2. 获取该企业该标准的企业目录配置
        var entConfig = await _db.GetOneAsync<StandardDirectoryConfig>(
            x => x.EnterpriseCode == enterpriseCode
                && x.StandardCode == stage.StandardCode
                && x.StageCode == stage.StageCode
                && x.IsValid == 1);
        if (entConfig == null) continue;

        // 3. 获取模板行（用于知道"应该有什么文件"）
        // ⚠️ 2026-09-30 勘误（R-14）：原稿写 `x.EnterpriseCode == null || x.EnterpriseCode == "YZH-STD-ENT"`，
        //    两个分支均已作废 —— ①列已改 NOT NULL DEFAULT ''，IS NULL 分支为死代码；
        //    ②禁止手写 "YZH-STD-ENT" 字面量，必须引用常量 YzhVirtualEnterprise.Code。
        var templateConfig = await _db.GetOneAsync<StandardDirectoryConfig>(
            x => x.OrgCode == /* workspace */ && x.StandardCode == stage.StandardCode
                && x.StageCode == stage.StageCode
                && x.EnterpriseCode == YzhVirtualEnterprise.Code);

        // 4. 获取模板文件清单
        var templateFiles = await _db.GetListAsync<StandardDirectoryFile>(
            x => x.ConfigCode == templateConfig.Code
                && x.EnterpriseCode == YzhVirtualEnterprise.Code
                && x.IsRequired == 1 && !x.IsDeleted);

        // 5. 获取企业实际上传文件
        var entFiles = await _db.GetListAsync<StandardDirectoryFile>(
            x => x.ConfigCode == entConfig.Code
                && x.EnterpriseCode == enterpriseCode
                && !x.IsDeleted);

        var summary = new StandardCheckSummary {
            StandardCode = stage.StandardCode,
            StandardName = /* 从 cert_iso_standard 查 */,
            TotalRequired = templateFiles.Count,
        };

        foreach (var tf in templateFiles)
        {
            var entFile = entFiles.FirstOrDefault(e => e.FileName == tf.FileName);
            if (entFile == null)
            {
                // 缺失
                summary.MissingFiles.Add(new MissingFile {
                    StandardFileCode = tf.Code,
                    FileName = tf.FileName,
                    FolderPath = /* 从文件夹树查 */,
                    StandardCode = stage.StandardCode,
                    IsRequired = tf.IsRequired
                });
                summary.MissingCount++;
            }
            else
            {
                summary.UploadedCount++;
                if (entFile.ExtractStatus == "completed")
                {
                    summary.ExtractedCount++;
                }
                else if (entFile.ExtractStatus == "failed" && entFile.MaxConfidence.HasValue
                         && entFile.MaxConfidence.Value < 0.5m)
                {
                    // 部分覆盖：已上传但提取质量差
                    summary.PartialFiles.Add(new PartialFile {
                        StandardFileCode = tf.Code,
                        FileName = tf.FileName,
                        FolderPath = /* 从文件夹树查 */,
                        MaxConfidence = entFile.MaxConfidence,
                        ExtractedFieldCount = /* 从 cert_extraction_result 聚合 */,
                        TotalFieldCount = /* 从 cert_doc_field_def 计数 */
                    });
                }
            }
        }

        result.Standards.Add(summary);
    }

    return result;
}
```

**调用时机**：
- 页面加载时自动调用（展示初始状态）
- 每次上传/替换/删除后重新调用（刷新状态）
- 用户手动点击「查看文件检查报告」时调用
- **任何后端服务随时可调**（如工作流节点读取企业资料完整性）

```
    ↓
更新 ExtractStatus + MaxConfidence
    ↓
页面自动刷新（或轮询）→ 重新调用 /FileCheck 更新状态
```

## 八、决策点

| ID | 决策点 | 建议 | 影响 |
|----|--------|------|------|
| E1 | 复用现有表 vs 新建企业表 | **复用现有表，加 EnterpriseCode 列** | 表结构最简 |
| E2 | 上传是否需要临时存储 | **不需要**（同桶 CopyObject 是元数据操作） | 简化流程 |
| E3 | 批量上传映射方式 | **自动推荐 + 人工确认** | 减少审核员工作量 |
| E4 | 提取触发 | **转换完成后自动触发** | 减少人工操作 |
| E5 | 历史版本保留 | **物理保留在 `_archive/` 下**（V3.1 勘误：原 `History/`；删除亦不物理销毁 MinIO 对象，见 06 册 02 号 §五-2） | 可回溯 |
| E6 | 存储抽象层 | **立即引入 IObjectStorageService 接口** | 为未来 OSS 迁移零成本切换 |
| E7 | 企业目录初始化时机 | **企业-阶段-标准关联建立时自动创建** | 审核员无需手工维护 |
| E8 | 文件检查调用时机 | **页面加载时自动 + 用户手动触发 + 后端随时调用** | 灵活性最高 |
